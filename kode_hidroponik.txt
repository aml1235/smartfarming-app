#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <PubSubClient.h>
#include <Preferences.h>
#include <DHT.h>
#include <ArduinoJson.h>

// ========================================
// CONFIGURATION
// ========================================
String WIFI_SSID = "iPhone";         
String WIFI_PASSWORD = "elma5321";   
const char* AP_SSID = "Lab Modern Alkomlek";     
const char* AP_PASSWORD = "prc_1077";       
const char* SECTOR_ID = "SEC-010";  

// ========================================
// MQTT CONFIGURATION (HiveMQ Cloud)
// ========================================
const char* MQTT_BROKER = "e8d98b46c66448469bd969da39a11bd8.s1.eu.hivemq.cloud"; // Ganti dengan URL HiveMQ Cloud Anda
const int MQTT_PORT = 8883; // Port TLS
const char* MQTT_USERNAME = "amel"; // Ganti dengan username HiveMQ Anda
const char* MQTT_PASSWORD = "amelpw123"; // Ganti dengan password HiveMQ Anda
const char* MQTT_CLIENT_ID = "ESP32_Hidroponik_SEC010";

// Topik khusus hidroponik agar tidak bentrok dengan kandang ayam
const char* TOPIC_SENSOR = "smartfarming/hydroponic/sensor/SEC-010";
const char* TOPIC_CMD = "smartfarming/hydroponic/cmd/SEC-010";

// ========================================
// TIMING & PINS
// ========================================
const unsigned long UPLOAD_INTERVAL = 20000;      
const unsigned long WIFI_RETRY_INTERVAL = 30000;  

#define DHT_PIN 4           
#define DHT_TYPE DHT22      
#define LDR_PIN 34          
#define TRIG_PIN 18         
#define ECHO_PIN 19         
#define RELAY_PIN 26        
#define LED_PIN 2           

DHT dht(DHT_PIN, DHT_TYPE);
WiFiClientSecure secureClient;
PubSubClient mqttClient(secureClient);
Preferences preferences;

// Variables
int wifiRetryCount = 0;
const int MAX_WIFI_RETRY = 5;  
bool apModeActive = false;

unsigned long lastUploadTime = 0;
unsigned long lastWiFiRetry = 0;

float temperature = 0.0;
float humidity = 0.0;
int lightLevel = 0;
float waterLevel = 0.0;
String pumpStatus = "ON";

bool wifiConnected = false;

// ========================================
// SETUP
// ========================================
void setup() {
  Serial.begin(115200);
  delay(1000);

  Serial.println("\n\n=== TAMTAMA ESP32 v4.0 (MQTT) ===");
  
  initializePins();
  dht.begin();
  secureClient.setInsecure(); // Abaikan sertifikat SSL untuk kemudahan

  mqttClient.setServer(MQTT_BROKER, MQTT_PORT);
  mqttClient.setCallback(mqttCallback);

  loadWiFiCredentials();
  connectWiFi();
}

// ========================================
// MAIN LOOP
// ========================================
void loop() {
  unsigned long currentMillis = millis();
  handleSerialCommands();

  if (WiFi.status() != WL_CONNECTED && !apModeActive) {
    wifiConnected = false;
    digitalWrite(LED_PIN, LOW);
    if (currentMillis - lastWiFiRetry >= WIFI_RETRY_INTERVAL) {
      connectWiFi();
      lastWiFiRetry = currentMillis;
    }
    delay(1000);
    return;
  }

  if (apModeActive) {
    // Mode AP tidak butuh MQTT
    digitalWrite(LED_PIN, (currentMillis / 200) % 2 == 0 ? HIGH : LOW);
    delay(10);
    return;
  }

  wifiConnected = true;

  if (!mqttClient.connected()) {
    reconnectMQTT();
  }
  mqttClient.loop(); // Handle incoming messages

  if (currentMillis - lastUploadTime >= UPLOAD_INTERVAL) {
    readAllSensors();
    publishSensorData();
    lastUploadTime = currentMillis;
  }

  digitalWrite(LED_PIN, (currentMillis / 1000) % 2 == 0 ? HIGH : LOW);
  delay(100);
}

// ========================================
// PIN INIT
// ========================================
void initializePins() {
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW); // Default ON
  pumpStatus = "ON";
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
}

// ========================================
// WIFI & MQTT
// ========================================
void connectWiFi() {
  Serial.println("\n[WiFi] Connecting to: " + WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.disconnect(true);
  delay(100);
  WiFi.begin(WIFI_SSID.c_str(), WIFI_PASSWORD.c_str());

  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 30) {
    delay(500); Serial.print("."); attempts++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n[WiFi] CONNECTED!");
    wifiConnected = true; wifiRetryCount = 0;
  } else {
    wifiConnected = false; wifiRetryCount++;
    if (wifiRetryCount >= MAX_WIFI_RETRY) startAPMode();
  }
}

void reconnectMQTT() {
  while (!mqttClient.connected() && WiFi.status() == WL_CONNECTED) {
    Serial.print("[MQTT] Connecting to Broker...");
    if (mqttClient.connect(MQTT_CLIENT_ID, MQTT_USERNAME, MQTT_PASSWORD)) {
      Serial.println(" CONNECTED!");
      mqttClient.subscribe(TOPIC_CMD);
      Serial.println("[MQTT] Subscribed to: " + String(TOPIC_CMD));
    } else {
      Serial.print(" FAILED, rc=");
      Serial.print(mqttClient.state());
      Serial.println(" try again in 5 seconds");
      delay(5000);
    }
  }
}

// ========================================
// SENSORS
// ========================================
void readAllSensors() {
  float newTemp = dht.readTemperature();
  float newHum = dht.readHumidity();
  if (!isnan(newTemp) && !isnan(newHum)) { temperature = newTemp; humidity = newHum; }

  int rawLDR = analogRead(LDR_PIN);
  lightLevel = constrain(map(rawLDR, 0, 4095, 100, 0), 0, 100);

  digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH, 30000);
  waterLevel = (duration == 0) ? 0.0 : (duration * 0.0343) / 2.0;
  if (waterLevel < 2.0 || waterLevel > 400.0) waterLevel = 0.0;
}

// ========================================
// PUBLISH SENSORS TO MQTT
// ========================================
void publishSensorData() {
  if (!mqttClient.connected()) return;
  Serial.print("[MQTT] Publishing data... ");

  StaticJsonDocument<256> doc;
  doc["temperature"] = round(temperature * 10) / 10.0;
  doc["humidity"] = round(humidity * 10) / 10.0;
  doc["lightLevel"] = lightLevel;
  doc["waterLevel"] = round(waterLevel * 10) / 10.0;
  doc["pumpStatus"] = pumpStatus;

  String payload; serializeJson(doc, payload);
  
  if (mqttClient.publish(TOPIC_SENSOR, payload.c_str())) {
    Serial.println("✅ OK");
  } else {
    Serial.println("❌ GAGAL");
  }
}

// ========================================
// MQTT CALLBACK (Menerima Perintah)
// ========================================
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (unsigned int i = 0; i < length; i++) {
    message += (char)payload[i];
  }

  Serial.println("\n[MQTT CMD] Pesan masuk di topik: " + String(topic));
  Serial.println("[MQTT CMD] Payload: " + message);

  StaticJsonDocument<256> doc;
  if (!deserializeJson(doc, message)) {
    String cmdStatus = doc["status"] | "";
    if (cmdStatus != "" && cmdStatus != pumpStatus) {
      Serial.println("[MQTT CMD] Mengeksekusi perintah: " + cmdStatus);
      setPumpStatus(cmdStatus);
      publishSensorData(); // Langsung publish status terbaru setelah berubah
    }
  }
}

void setPumpStatus(String status) {
  status.toUpperCase();
  if (status == "ON") { digitalWrite(RELAY_PIN, LOW); pumpStatus = "ON"; } 
  else if (status == "OFF") { digitalWrite(RELAY_PIN, HIGH); pumpStatus = "OFF"; }
}

// ========================================
// TOOLS & UTILS
// ========================================
void handleSerialCommands() {
  if (!Serial.available()) return;
  String cmd = Serial.readStringUntil('\n');
  cmd.trim(); cmd.toLowerCase();
  if (cmd == "on") setPumpStatus("ON");
  else if (cmd == "off") setPumpStatus("OFF");
  else if (cmd == "restart") ESP.restart();
  else if (cmd == "apmode") startAPMode();
}

void loadWiFiCredentials() {
  preferences.begin("wifi", false);
  String s = preferences.getString("ssid", "");
  String p = preferences.getString("password", "");
  if (s.length() > 0) { WIFI_SSID = s; WIFI_PASSWORD = p; }
  preferences.end();
}

void startAPMode() {
  WiFi.disconnect(true); delay(100);
  WiFi.mode(WIFI_AP); WiFi.softAP(AP_SSID, AP_PASSWORD);
  apModeActive = true;
}

void exitAPMode() {
  WiFi.softAPdisconnect(true); WiFi.mode(WIFI_STA);
  apModeActive = false; wifiConnected = true; wifiRetryCount = 0;
}
