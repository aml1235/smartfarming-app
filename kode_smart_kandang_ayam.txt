/* ============================================================================
   HELPER TIMER: parsing "HH:MM"  <->  menit
   ============================================================================ */
bool parseHHMM(const String& s, uint16_t &outMinutes) {
  // Format valid: "H:MM", "HH:MM" (panjang 4 atau 5)
  int colon = s.indexOf(':');
  if (colon < 1 || colon > 2) return false;
  if (s.length() - colon - 1 != 2) return false;

  for (unsigned int i = 0; i < s.length(); i++) {
    if (i == (unsigned int)colon) continue;
    if (!isdigit(s.charAt(i))) return false;
  }

  int h = s.substring(0, colon).toInt();
  int m = s.substring(colon + 1).toInt();
  if (h < 0 || h > 23 || m < 0 || m > 59) return false;

  outMinutes = (uint16_t)(h * 60 + m);
  return true;
}

void minToStr(uint16_t v, char* out) {          // out minimal char[6]
  sprintf(out, "%02u:%02u", v / 60, v % 60);
}

// Cek apakah "nowMin" berada di dalam jendela onMin -> offMin.
// Mendukung jendela yang melewati tengah malam (contoh: 18:00 -> 06:00).
bool inWindow(uint16_t nowMin, uint16_t onMin, uint16_t offMin) {
  if (onMin == offMin) return false;                 // jendela kosong = nonaktif
  if (onMin < offMin)  return (nowMin >= onMin && nowMin < offMin);
  return (nowMin >= onMin || nowMin < offMin);       // lewat tengah malam
}

  lampOnMin    = prefs.getUShort("lampOn",   18 * 60);
  lampOffMin   = prefs.getUShort("lampOff",   6 * 60);

  conv2Enabled = prefs.getBool  ("cv2En",    true);
  convRunSec   = prefs.getUShort("cvRun",    10);
  convPauseSec = prefs.getUShort("cvPause",  3);
  convSpeedPct = prefs.getUChar ("cvSpeed",  100);
  prefs.end();

  // Jaga-jaga kalau NVS pernah terisi nilai aneh
  if (lampOnMin  > 1439) lampOnMin  = 18 * 60;
  if (lampOffMin > 1439) lampOffMin =  6 * 60;
  if (conv1OnMin > 1439) conv1OnMin = 6 * 60;
  if (conv2OnMin > 1439) conv2OnMin = 18 * 60;
  if (convRunSec < 1 || convRunSec > 300) convRunSec = 10;

  if (convPauseSec < 1 || convPauseSec > 300) convPauseSec = 3;
  if (convSpeedPct < 20 || convSpeedPct > 100) convSpeedPct = 100;

  char b1[6], b2[6], b3[6], b5[6];
  minToStr(lampOnMin, b1);   minToStr(lampOffMin, b2);
  minToStr(conv1OnMin, b3);  minToStr(conv2OnMin, b5);
  Serial.println(F("[NVS] Jadwal dimuat:"));
  Serial.print(F("      Lampu     : ")); Serial.print(b1); Serial.print(F(" -> ")); Serial.println(b2);
  Serial.print(F("      Conveyor 1: ")); Serial.println(b3);
  Serial.print(F("      Conveyor 2: ")); Serial.print(b5);
  Serial.println(conv2Enabled ? F("  [AKTIF]") : F("  [NONAKTIF]"));
  Serial.print(F("      Siklus    : maju/mundur ")); Serial.print(convRunSec);
  Serial.print(F(" dtk, jeda ")); Serial.print(convPauseSec);
  Serial.print(F(" dtk, kecepatan ")); Serial.print(convSpeedPct); Serial.println(F(" %"));
}

/* ============================================================================
   NTP - wajib sebelum TLS
   ============================================================================ */
void syncNTP() {
  Serial.println(F("[NTP] Syncing time..."));
  configTime(7 * 3600, 0, "pool.ntp.org", "time.google.com"); // GMT+7 (WIB)

  time_t now = time(nullptr);
  int retries = 0;
  while (now < 1700000000 && retries < 20) {
    delay(500);
    now = time(nullptr);
    retries++;
    Serial.print(F("."));
  }
  Serial.println();

  if (now < 1700000000) {
    Serial.println(F("[NTP] FAILED! Time not synced. TLS may fail."));
  } else {
    struct tm timeinfo;
    localtime_r(&now, &timeinfo);
    Serial.print(F("[NTP] OK! Current time: "));
    Serial.println(asctime(&timeinfo));
  }
}

/* ============================================================================
   RTC
   ============================================================================ */
bool isRTCValid(DateTime dt) {
  if (dt.year() < 2024 || dt.year() > 2035) return false;
  if (dt.month() < 1 || dt.month() > 12) return false;
  if (dt.day() < 1 || dt.day() > 31) return false;
  if (dt.hour() > 23) return false;
  if (dt.minute() > 59) return false;
  if (dt.second() > 59) return false;
  return true;
}

// NTP dianggap sah kalau epoch sudah melewati akhir 2023
bool ntpValid() { return time(nullptr) > 1700000000; }

/*
   Sumber waktu utama adalah NTP, bukan RTC.

   Yang sering disalahpahami: ESP32 TIDAK kehilangan jam saat WiFi putus.
   Begitu NTP menyetel waktu sistem sekali, jam internal terus berjalan
   sendiri. WiFi hanya dibutuhkan untuk sinkronisasi berkala. Jadi memakai
   NTP justru lebih tahan gangguan daripada RTC yang modulnya bermasalah.

   RTC tetap dipakai sebagai cadangan untuk kondisi listrik mati lalu
   menyala kembali sementara WiFi belum tersedia.
*/
DateTime getSafeTime() {
  if (ntpValid()) {
    struct tm t;
    if (getLocalTime(&t, 10)) {
      timeFromNTP = true;
      return DateTime(t.tm_year + 1900, t.tm_mon + 1, t.tm_mday,
                      t.tm_hour, t.tm_min, t.tm_sec);
    }
  }

  timeFromNTP = false;
  if (!rtcAvailable) return rtcLastValid + TimeSpan(millis() / 1000);
  DateTime now = rtc.now();
  if (!isRTCValid(now)) {
    Serial.println(F("[RTC] Tidak valid, dipakai waktu kompilasi"));
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    now = rtc.now();
  }
  rtcLastValid = now;
  return now;
}

// Selaraskan RTC dari NTP secara berkala supaya cadangannya ikut akurat
void rtcSyncTask(unsigned long ms) {
  if (!rtcAvailable || !ntpValid()) return;
  if (prevRtcSyncMillis != 0 && ms - prevRtcSyncMillis < RTC_SYNC_INTERVAL) return;
  prevRtcSyncMillis = ms;

  struct tm t;
  if (!getLocalTime(&t, 10)) return;
  rtc.adjust(DateTime(t.tm_year + 1900, t.tm_mon + 1, t.tm_mday,
                      t.tm_hour, t.tm_min, t.tm_sec));
  Serial.println(F("[RTC] Diselaraskan dari NTP"));
}

/* ============================================================================
   LCD - versi anti karakter corrupt
   ============================================================================ */
char lcdPrev[LCD_ROWS][LCD_COLS + 1];

uint8_t scanI2CAddress() {
  const uint8_t addresses[] = {0x27, 0x3F};
  for (uint8_t i = 0; i < 2; i++) {
    Wire.beginTransmission(addresses[i]);
    if (Wire.endTransmission() == 0) return addresses[i];
  }
  for (uint8_t addr = 8; addr < 120; addr++) {
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) return addr;
  }
  return 0;
}

bool i2cPing(uint8_t addr) {
  Wire.beginTransmission(addr);
  return (Wire.endTransmission() == 0);
}

void lcdForceRedraw() {
  for (uint8_t r = 0; r < LCD_ROWS; r++) lcdPrev[r][0] = '\0';
}

// Tulis satu baris penuh 20 karakter dalam SATU panggilan print().
// Ini kunci fix corrupt: LCD tidak lagi menerima banyak burst I2C pendek
// yang gampang terpotong saat relay switching.
void lcdWriteLine(uint8_t row, const char* text) {
  if (!lcdAvailable || lcd == nullptr || row >= LCD_ROWS) return;

  char buf[LCD_COLS + 1];
  snprintf(buf, sizeof(buf), "%-*s", LCD_COLS, text);   // pad kanan sampai 20 char
  buf[LCD_COLS] = '\0';

  if (strcmp(buf, lcdPrev[row]) == 0) return;           // tidak berubah -> skip
  strcpy(lcdPrev[row], buf);

  lcd->setCursor(0, row);
  delayMicroseconds(120);        // beri waktu HD44780 memproses perintah alamat
  lcd->print(buf);
}

bool initLCD() {
  lcdAddress = scanI2CAddress();
  if (lcdAddress == 0) {
    Serial.println(F("[LCD] NOT DETECTED"));
    return false;
  }

  if (lcd == nullptr) lcd = new LiquidCrystal_I2C(lcdAddress, LCD_COLS, LCD_ROWS);

  lcd->init();
  delay(100);                      // HD44780 butuh waktu setelah init
  Wire.setClock(I2C_CLOCK_LCD);    // library init() sempat reset clock -> set ulang
  lcd->backlight();
  delay(50);
  lcd->clear();
  delay(50);                       // clear butuh ~1.6 ms, beri margin

  lcdForceRedraw();
  Serial.print(F("[LCD] OK at 0x")); Serial.println(lcdAddress, HEX);
  return true;
}

// Kalau LCD hilang dari bus (kena spike relay), init ulang otomatis
void lcdHealthCheck(unsigned long currentMillis) {
  if (!lcdAvailable) return;
  if (currentMillis - prevLcdCheckMillis < LCD_CHECK_INTERVAL) return;
  prevLcdCheckMillis = currentMillis;

  if (!i2cPing(lcdAddress)) {
    Serial.println(F("[LCD] Tidak merespon! Init ulang..."));
    lcd->init();
    delay(100);
    Wire.setClock(I2C_CLOCK_LCD);
    lcd->backlight();
    delay(50);
    lcdForceRedraw();
  }
}

void updateLCD(unsigned long currentMillis) {
  if (!lcdAvailable || lcd == nullptr) return;
  if (currentMillis - prevLcdMillis < LCD_INTERVAL) return;
  prevLcdMillis = currentMillis;

  DateTime now = getSafeTime();
  char line[LCD_COLS + 1];

  // CATATAN: jangan pakai "%f" di snprintf. Sebagian build ESP32 memakai
  // newlib "nano format" yang tidak mendukung float, hasilnya jadi kosong/0.
  // dtostrf() adalah rutin Arduino dan selalu bekerja.
  char tStr[10], hStr[10];
  if (dhtValid) {
    dtostrf(temperature, 0, 1, tStr);
    dtostrf(humidity,    0, 0, hStr);
  } else {
    strcpy(tStr, "--");         // DHT belum pernah berhasil dibaca
    strcpy(hStr, "--");
  }
  snprintf(line, sizeof(line), "Suhu  :%sC Hum:%s%%", tStr, hStr);
  lcdWriteLine(0, line);

  snprintf(line, sizeof(line), "Amonia:%-4d  Air:%-3d%%", mq135Value, waterLevelPercent);
  lcdWriteLine(1, line);

  snprintf(line, sizeof(line), "Conv  :%-4s Pump:%-3s",
           convPhaseName(), pompaStatus ? "ON" : "OFF");
  lcdWriteLine(2, line);

  snprintf(line, sizeof(line), "%-6s %02d:%02d Lmp:%-3s",
           NAMA_HARI[now.dayOfTheWeek()], now.hour(), now.minute(),
           lampStatus ? "ON" : "OFF");
  lcdWriteLine(3, line);
}

/* ============================================================================
   MQTT CALLBACK
   ============================================================================ */
void handleTimerConfig(const char* topic, const String& message) {
  uint16_t v = 0;
  bool changed = false;

  if (strcmp(topic, TOPIC_CFG_CONV2EN) == 0) {
    bool en = (message == "1" || message == "ON" || message == "true");
    if (en != conv2Enabled) { conv2Enabled = en; changed = true; }
  }
  else if (strcmp(topic, TOPIC_CFG_CONVRUN) == 0) {
    int d = message.toInt();
    if (d < 1 || d > 300) { Serial.println(F("[CONFIG] Durasi maju/mundur harus 1-300 detik")); return; }
    if ((uint16_t)d != convRunSec) { convRunSec = d; changed = true; }
  }
  else if (strcmp(topic, TOPIC_CFG_CONVPAUSE) == 0) {
    int d = message.toInt();
    if (d < 1 || d > 300) { Serial.println(F("[CONFIG] Jeda harus 1-300 detik")); return; }
    if ((uint16_t)d != convPauseSec) { convPauseSec = d; changed = true; }
  }
  else if (strcmp(topic, TOPIC_CFG_CONVSPEED) == 0) {
    int d = message.toInt();
    // Di bawah 20% motor DC biasanya hanya mendengung tanpa berputar
    if (d < 20 || d > 100) { Serial.println(F("[CONFIG] Kecepatan harus 20-100 persen")); return; }
    if ((uint8_t)d != convSpeedPct) { convSpeedPct = d; changed = true; }
  }
  else if (!parseHHMM(message, v)) {
    Serial.print(F("[CONFIG] Format jam tidak valid: ")); Serial.println(message);
    return;
  }
  else if (strcmp(topic, TOPIC_CFG_LAMPON) == 0)   { if (v != lampOnMin)   { lampOnMin = v;   changed = true; } }
  else if (strcmp(topic, TOPIC_CFG_LAMPOFF) == 0)  { if (v != lampOffMin)  { lampOffMin = v;  changed = true; } }
  else if (strcmp(topic, TOPIC_CFG_CONVON) == 0)   { if (v != conv1OnMin) { conv1OnMin = v; changed = true; } }
  else if (strcmp(topic, TOPIC_CFG_CONV2ON) == 0)  { if (v != conv2OnMin) { conv2OnMin = v; changed = true; } }

  if (changed) {
    Serial.print(F("[CONFIG] Jadwal diperbarui via ")); Serial.println(topic);
    timerConfigDirty = true;          // simpan & publish dikerjakan di loop()
  }
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (unsigned int i = 0; i < length; i++) message += (char)payload[i];
  message.trim();
  Serial.print(F("[MQTT] RX [")); Serial.print(topic); Serial.print(F("]: ")); Serial.println(message);

  if (strncmp(topic, "smartcoop/config/", 17) == 0) {
    handleTimerConfig(topic, message);
    return;
  }

  if (strcmp(topic, TOPIC_CTRL_LAMP) == 0) {
    lampManualState = (message == "1" || message == "ON" || message == "true");
    if (!lampAutoMode) {
      lampStatus = lampManualState;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      mqtt.publish(TOPIC_LAMP, lampStatus ? "1" : "0", true);
    }
  }
  else if (strcmp(topic, TOPIC_CTRL_CONVJOG) == 0) {
    if (message == "fwd" || message == "maju")       convStartJog(true);
    else if (message == "rev" || message == "mundur") convStartJog(false);
    else convStopNow("jog dilepas");
  }
  else if (strcmp(topic, TOPIC_CTRL_CONVEYOR) == 0) {
    if (message == "1" || message == "ON" || message == "true") convStartCycle("manual");
    else convStopNow("perintah manual");
  }
  else if (strcmp(topic, TOPIC_CTRL_POMPA) == 0) {
    pompaManualState = (message == "1" || message == "ON" || message == "true");
    if (!pompaAutoMode) {
      pompaStatus = pompaManualState;
      digitalWrite(RELAY_POMPA, pompaStatus ? LOW : HIGH);
      mqtt.publish(TOPIC_POMPA, pompaStatus ? "1" : "0", true);
      Serial.print(F("[POMPA] Manual -> ")); Serial.println(pompaStatus ? F("ON") : F("OFF"));
    } else {
      Serial.println(F("[POMPA] Diabaikan: masih mode AUTO. Matikan mode auto dulu."));
    }
  }
  else if (strcmp(topic, TOPIC_CTRL_POMPAAUTO) == 0) {
    pompaAutoMode = (message == "1" || message == "ON" || message == "true");
    if (pompaAutoMode) {
      // Kembali ke auto: selaraskan state machine dengan kondisi relay saat ini
      pumpAutoState = pompaStatus ? PUMP_AUTO_ON : PUMP_AUTO_IDLE;
      Serial.println(F("[POMPA] Mode AUTO"));
    } else {
      pompaManualState = pompaStatus;   // mulai manual dari kondisi sekarang
      Serial.println(F("[POMPA] Mode MANUAL"));
    }
    mqtt.publish(TOPIC_POMPAAUTO, pompaAutoMode ? "1" : "0", true);
  }
  else if (strcmp(topic, TOPIC_CTRL_LAMPAUTO) == 0) {
    lampAutoMode = (message == "1" || message == "ON" || message == "true");
    if (!lampAutoMode) {
      lampStatus = lampManualState;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      mqtt.publish(TOPIC_LAMP, lampStatus ? "1" : "0", true);
    }
    mqtt.publish(TOPIC_LAMPAUTO, lampAutoMode ? "1" : "0", true);
  }
}

/* ============================================================================
   MQTT RECONNECT
   ============================================================================ */
bool reconnectMQTT() {
  Serial.println(F("\n========================================"));
  Serial.println(F("[MQTT] Attempting connection..."));
  Serial.print(F("[MQTT] Server: ")); Serial.print(mqtt_server);
  Serial.print(F(":")); Serial.println(mqtt_port);

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println(F("[MQTT] ABORT: WiFi not connected!"));
    Serial.print(F("[WIFI] Status: ")); Serial.println(wifiStateString(WiFi.status()));
    return false;
  }

  time_t now = time(nullptr);
  if (now < 1700000000) {
    Serial.println(F("[MQTT] WARNING: NTP time not synced yet! TLS may fail."));
  }

  if (mqtt.connect(client_id, mqtt_user, mqtt_pass)) {
    Serial.println(F("[MQTT] >>> CONNECTED SUCCESS! <<<"));

    mqtt.subscribe(TOPIC_CTRL_LAMP);
    mqtt.subscribe(TOPIC_CTRL_CONVEYOR);
    mqtt.subscribe(TOPIC_CTRL_CONVJOG);
    mqtt.subscribe(TOPIC_CTRL_POMPA);
    mqtt.subscribe(TOPIC_CTRL_LAMPAUTO);
    mqtt.subscribe(TOPIC_CTRL_POMPAAUTO);
    mqtt.subscribe(TOPIC_CFG_LAMPON);
    mqtt.subscribe(TOPIC_CFG_LAMPOFF);
    mqtt.subscribe(TOPIC_CFG_CONVON);
    mqtt.subscribe(TOPIC_CFG_CONV2ON);
    mqtt.subscribe(TOPIC_CFG_CONV2EN);
    mqtt.subscribe(TOPIC_CFG_CONVRUN);
    mqtt.subscribe(TOPIC_CFG_CONVPAUSE);
    mqtt.subscribe(TOPIC_CFG_CONVSPEED);
    Serial.println(F("[MQTT] Subscribed: control + config"));

    mqtt.publish(TOPIC_SYSTEM, "online", true);
    publishTimerConfig();
    return true;
  }

  int state = mqtt.state();
  Serial.print(F("[MQTT] FAILED! Code: ")); Serial.print(state);
  Serial.print(F(" = ")); Serial.println(mqttStateString(state));
  if (state == -2) Serial.println(F("[HINT] -2 CONN_FAILED: Cek host, port, WiFi, atau TLS cert"));
  if (state == -4) Serial.println(F("[HINT] -4 TIMEOUT: Server tidak merespon, cek firewall/port"));
  if (state == 4)  Serial.println(F("[HINT] 4 BAD_CREDENTIALS: Username atau password SALAH!"));
  if (state == 5)  Serial.println(F("[HINT] 5 UNAUTHORIZED: Password salah atau client ID diblokir"));
  if (state == 2)  Serial.println(F("[HINT] 2 BAD_CLIENT_ID: Client ID terlalu panjang atau duplikat"));
  Serial.println(F("========================================"));
  return false;
}

/* ============================================================================
   SENSOR
   ============================================================================ */
void readSensor(unsigned long currentMillis) {
  if (currentMillis - prevSensorMillis >= SENSOR_INTERVAL) {
    prevSensorMillis = currentMillis;
    float h = dht.readHumidity();
    float t = dht.readTemperature();
    if (isnan(h) || isnan(t)) {
      Serial.println(F("[SENSOR] DHT22 error! (cek pin 13, resistor pull-up 10k, supply 5V)"));
    } else {
      temperature = t; humidity = h - 40; dhtValid = true;
    }
    mq135Value = analogRead(MQ135_PIN);
    mq135Voltage = mq135Value * (3.3 / 4095.0);
    if (mq135Value < 50) Serial.println(F("[SENSOR] MQ135 low! Check wiring."));
  }
}

void readWaterSensor(unsigned long currentMillis) {
  if (currentMillis - prevWaterReadMillis >= WATER_READ_INTERVAL) {
    prevWaterReadMillis = currentMillis;
    int raw = analogRead(WATER_PIN);
    waterSampleSum -= waterSamples[waterSampleIndex];
    waterSamples[waterSampleIndex] = raw;
    waterSampleSum += raw;
    waterSampleIndex++;
    if (waterSampleIndex >= WATER_SAMPLE_COUNT) waterSampleIndex = 0;
    waterAvgADC = waterSampleSum / WATER_SAMPLE_COUNT;
  }
}

void calculateWaterLevel() {
  waterVoltage = waterAvgADC * (3.3 / 4095.0);
  waterLevelPercent = map(waterAvgADC, 100, 2500, 0, 100);
  waterLevelPercent = constrain(waterLevelPercent, 0, 100);
}

/* ============================================================================
   WIFI TASK
   ============================================================================ */
void wifiTask(unsigned long currentMillis) {
  if (currentMillis - prevWifiMillis >= WIFI_INTERVAL) {
    prevWifiMillis = currentMillis;
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println(F("[WIFI] Reconnecting..."));
      WiFi.disconnect(); WiFi.begin(ssid, pass);
    }
  }
}

/* ============================================================================
   LAMPU OTOMATIS - pakai jadwal dari NVS/MQTT
   ============================================================================ */
void lampAutomation(unsigned long currentMillis) {
  static bool lastAutoMode = true;

  if (lampAutoMode) {
    if (!lastAutoMode) lastAutoMode = true;
    DateTime now = getSafeTime();
    uint16_t nowMin = now.hour() * 60 + now.minute();
    bool shouldLampOn = inWindow(nowMin, lampOnMin, lampOffMin);

    if (shouldLampOn != lampStatus) {
      lampStatus = shouldLampOn;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      if (mqtt.connected()) mqtt.publish(TOPIC_LAMP, lampStatus ? "1" : "0", true);
      Serial.print(F("[LAMPU] Jadwal -> ")); Serial.println(lampStatus ? F("ON") : F("OFF"));
    }
  } else {
    if (lastAutoMode) {
      lastAutoMode = false;
      lampStatus = lampManualState;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      if (mqtt.connected()) mqtt.publish(TOPIC_LAMP, lampStatus ? "1" : "0", true);
    } else if (lampStatus != lampManualState) {
      lampStatus = lampManualState;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      if (mqtt.connected()) mqtt.publish(TOPIC_LAMP, lampStatus ? "1" : "0", true);
    }
  }
}

/* ============================================================================
   CONVEYOR MOTOR DC - driver H-bridge MOSFET (PWM + DIR)
   ----------------------------------------------------------------------------
   Satu siklus: MAJU -> JEDA -> (dead-time) -> MUNDUR -> berhenti.
   Arah TIDAK PERNAH diubah saat PWM masih aktif. Mengubah DIR sementara
   motor berputar membuat kedua sisi jembatan konduksi sesaat (shoot-through)
   dan itulah cara tercepat membakar MOSFET.
   ============================================================================ */
void motorPwmWrite(uint8_t duty) {
#if ESP_ARDUINO_VERSION_MAJOR >= 3
  ledcWrite(MOTOR_PWM_PIN, duty);
#else
  ledcWrite(MOTOR_PWM_CH, duty);
#endif
}

void motorInit() {
  pinMode(MOTOR_DIR_PIN, OUTPUT);
  digitalWrite(MOTOR_DIR_PIN, LOW);
#if ESP_ARDUINO_VERSION_MAJOR >= 3
  ledcAttach(MOTOR_PWM_PIN, MOTOR_PWM_FREQ, MOTOR_PWM_RES);
#else
  ledcSetup(MOTOR_PWM_CH, MOTOR_PWM_FREQ, MOTOR_PWM_RES);
  ledcAttachPin(MOTOR_PWM_PIN, MOTOR_PWM_CH);
#endif
  motorPwmWrite(0);
  Serial.println(F("[MOTOR] Driver siap, motor berhenti"));
}

void motorStop() {
  motorPwmWrite(0);
}

// forward = true -> DIR LOW, false -> DIR HIGH.
// Kalau motor berputar terbalik dari yang Anda harapkan, tukar saja
// kedua kabel motor di terminal driver, jangan ubah kode.
void motorRun(bool forward) {
  digitalWrite(MOTOR_DIR_PIN, forward ? LOW : HIGH);
  uint8_t duty = (uint16_t)convSpeedPct * 255 / 100;
  motorPwmWrite(duty);
}

const char* convPhaseName() {
  switch (convPhase) {
    case CONV_FWD:
    case CONV_JOG_FWD:  return "Maju";
    case CONV_PAUSE:
    case CONV_DEADTIME: return "Jeda";
    case CONV_REV:
    case CONV_JOG_REV:  return "Mndr";
    default:            return "Diam";
  }
}

bool convIsJogging() {
  return convPhase == CONV_JOG_FWD || convPhase == CONV_JOG_REV;
}

void publishConvStatus() {
  if (!mqtt.connected()) return;
  mqtt.publish(TOPIC_CONVEYOR, convPhase == CONV_IDLE ? "0" : "1", true);
  mqtt.publish(TOPIC_CONVPHASE, convPhaseName(), true);
}

void convStartCycle(const char* sumber) {
  if (convPhase != CONV_IDLE) {
    Serial.println(F("[CONVEYOR] Diabaikan: siklus sedang berjalan"));
    return;
  }
  convPhase = CONV_FWD;
  convAfterDead = CONV_REV;
  convPhaseMillis = millis();
  motorRun(true);
  publishConvStatus();
  Serial.print(F("[CONVEYOR] Siklus mulai, sumber: ")); Serial.println(sumber);
}

// Jog manual. Motor SELALU dihentikan dulu lalu melewati dead-time sebelum
// berputar, apa pun kondisi sebelumnya. Membalik DIR sementara motor masih
// berputar membuat kedua sisi H-bridge konduksi bersamaan dan MOSFET terbakar.
void convStartJog(bool forward) {
  ConvPhase target = forward ? CONV_JOG_FWD : CONV_JOG_REV;

  // Tombol arah yang sama ditekan lagi -> cukup ulang hitungan pengaman
  if (convPhase == target) {
    convPhaseMillis = millis();
    Serial.println(F("[CONVEYOR] Jog diperpanjang 10 detik"));
    return;
  }

  motorStop();
  convAfterDead = target;
  convPhase = CONV_DEADTIME;
  convPhaseMillis = millis();
  publishConvStatus();
  Serial.print(F("[CONVEYOR] Jog manual "));
  Serial.println(forward ? F("MAJU") : F("MUNDUR"));
}

void convStopNow(const char* alasan) {
  motorStop();
  convPhase = CONV_IDLE;
  convAfterDead = CONV_IDLE;
  convPhaseMillis = millis();
  publishConvStatus();
  Serial.print(F("[CONVEYOR] Berhenti: ")); Serial.println(alasan);
}

// PENTING: waktu diambil SEGAR di sini, bukan dioper dari loop().
// Kalau memakai nilai millis() yang diambil di awal loop(), sementara
// convPhaseMillis dicatat belakangan di dalam callback MQTT, maka
// (ms - convPhaseMillis) menjadi negatif. Pada unsigned long hasilnya
// membesar jadi ~4,29 miliar, dan setiap fase langsung dianggap habis.
void conveyorStateMachine() {
  unsigned long ms = millis();

  // Perbandingan bertanda: aman walau convPhaseMillis sempat mendahului ms
  #define CONV_ELAPSED(d) ((long)(ms - convPhaseMillis) >= (long)(d))

  switch (convPhase) {
    case CONV_IDLE:
      break;

    case CONV_FWD:
      if (CONV_ELAPSED(convRunSec * 1000UL)) {
        motorStop();                       // PWM nol DULU, arah belum diubah
        convPhase = CONV_PAUSE;
        convPhaseMillis = ms;
        publishConvStatus();
        Serial.println(F("[CONVEYOR] Maju selesai, jeda"));
      }
      break;

    case CONV_PAUSE:
      if (CONV_ELAPSED(convPauseSec * 1000UL)) {
        convPhase = CONV_DEADTIME;
        convPhaseMillis = ms;
      }
      break;

    case CONV_DEADTIME:
      // Jaminan terakhir: motor sudah benar-benar diam sebelum DIR dibalik
      if (CONV_ELAPSED(MOTOR_DEADTIME_MS)) {
        convPhase = convAfterDead;
        convPhaseMillis = ms;
        switch (convPhase) {
          case CONV_REV:
            motorRun(false);
            Serial.print(F("[CONVEYOR] Mundur ("));
            Serial.print(convRunSec * 1000UL + CONV_REV_EXTRA_MS);
            Serial.println(F(" ms)"));
            break;
          case CONV_JOG_FWD: motorRun(true);  break;
          case CONV_JOG_REV: motorRun(false); break;
          default:           motorStop();     break;
        }
        publishConvStatus();
      }
      break;

    case CONV_REV:
      if (CONV_ELAPSED(convRunSec * 1000UL + CONV_REV_EXTRA_MS)) {
        convStopNow("siklus selesai");
      }
      break;

    case CONV_JOG_FWD:
    case CONV_JOG_REV:
      // Pengaman: jog manual tidak pernah berjalan lebih dari 10 detik,
      // walau dashboard tertutup atau WiFi putus setelah tombol ditekan
      if (CONV_ELAPSED(CONV_JOG_MAX_MS)) {
        convStopNow("pengaman jog 10 detik");
      }
      break;
  }
  #undef CONV_ELAPSED
}

// Jadwal: siklus dipicu sekali pada jam yang ditentukan, maksimal 1x per hari
void conveyorSchedule() {
  if (convIsJogging()) return;    // jangan ganggu kendali manual
  DateTime now = getSafeTime();
  uint16_t nowMin = now.hour() * 60 + now.minute();
  int today = now.day();

  if (nowMin == conv1OnMin && convLastTrigDay1 != today) {
    convLastTrigDay1 = today;
    convStartCycle("jadwal 1");
  }
  if (conv2Enabled && nowMin == conv2OnMin && convLastTrigDay2 != today) {
    convLastTrigDay2 = today;
    convStartCycle("jadwal 2");
  }
}

/* ============================================================================
   POMPA OTOMATIS
   ============================================================================ */
void pumpAutomation(unsigned long currentMillis) {
  if (!pompaAutoMode) {
    // Mode manual: pompa sepenuhnya mengikuti perintah dari web
    if (pompaStatus != pompaManualState) {
      pompaStatus = pompaManualState;
      digitalWrite(RELAY_POMPA, pompaStatus ? LOW : HIGH);
      if (mqtt.connected()) mqtt.publish(TOPIC_POMPA, pompaStatus ? "1" : "0", true);
    }
    return;
  }

  switch (pumpAutoState) {
    case PUMP_AUTO_IDLE:
      if (waterAvgADC <= 100) {
        pumpAutoState = PUMP_AUTO_CONFIRM_LOW;
        pumpConfirmMillis = currentMillis;
        Serial.println(F("[PUMP] Low water! Confirming ON..."));
      }
      break;
    case PUMP_AUTO_CONFIRM_LOW:
      if (waterAvgADC > 100) {
        pumpAutoState = PUMP_AUTO_IDLE;
        Serial.println(F("[PUMP] Cancel ON (level rose)"));
      } else if (currentMillis - pumpConfirmMillis >= PUMP_CONFIRM_DELAY) {
        pompaStatus = true;
        digitalWrite(RELAY_POMPA, LOW);
        pumpAutoState = PUMP_AUTO_ON;
        Serial.println(F("[PUMP] CONFIRMED ON"));
        if (mqtt.connected()) mqtt.publish(TOPIC_POMPA, "1", true);
      }
      break;
    case PUMP_AUTO_ON:
      if (waterAvgADC >= 2500) {
        pumpAutoState = PUMP_AUTO_CONFIRM_HIGH;
        pumpConfirmMillis = currentMillis;
        Serial.println(F("[PUMP] High water! Confirming OFF..."));
      }
      break;
    case PUMP_AUTO_CONFIRM_HIGH:
      if (waterAvgADC < 2500) {
        pumpAutoState = PUMP_AUTO_ON;
        Serial.println(F("[PUMP] Cancel OFF (level dropped)"));
      } else if (currentMillis - pumpConfirmMillis >= PUMP_CONFIRM_DELAY) {
        pompaStatus = false;
        digitalWrite(RELAY_POMPA, HIGH);
        pumpAutoState = PUMP_AUTO_IDLE;
        Serial.println(F("[PUMP] CONFIRMED OFF"));
        if (mqtt.connected()) mqtt.publish(TOPIC_POMPA, "0", true);
      }
      break;
  }
}

/* ============================================================================
   PUBLISH
   ============================================================================ */
void publishMQTT(unsigned long currentMillis) {
  if (currentMillis - prevPublishMillis < PUBLISH_INTERVAL) return;
  prevPublishMillis = currentMillis;

  if (!mqtt.connected()) {
    Serial.println(F("[PUB] SKIP: MQTT not connected"));
    return;
  }

  char buf[32];
  dtostrf(temperature, 1, 1, buf);   mqtt.publish(TOPIC_TEMP, buf);
  dtostrf(humidity, 1, 1, buf);      mqtt.publish(TOPIC_HUM, buf);
  itoa(mq135Value, buf, 10);         mqtt.publish(TOPIC_MQ135, buf);
  dtostrf(mq135Voltage, 1, 3, buf);  mqtt.publish(TOPIC_MQ135_VOLT, buf);

  mqtt.publish(TOPIC_LAMP,     lampStatus     ? "1" : "0", true);
  mqtt.publish(TOPIC_CONVEYOR, convPhase == CONV_IDLE ? "0" : "1", true);
  mqtt.publish(TOPIC_CONVPHASE, convPhaseName(), true);
  mqtt.publish(TOPIC_POMPA,    pompaStatus    ? "1" : "0", true);
  mqtt.publish(TOPIC_LAMPAUTO, lampAutoMode   ? "1" : "0", true);
  mqtt.publish(TOPIC_POMPAAUTO, pompaAutoMode ? "1" : "0", true);

  DateTime now = getSafeTime();
  char timeStr[30];
  sprintf(timeStr, "%s, %02d:%02d:%02d",
          NAMA_HARI[now.dayOfTheWeek()], now.hour(), now.minute(), now.second());
  mqtt.publish(TOPIC_TIME, timeStr, true);
}

void publishWaterData(unsigned long currentMillis) {
  static unsigned long prevWaterPublishMillis = 0;
  if (currentMillis - prevWaterPublishMillis < PUBLISH_INTERVAL) return;
  prevWaterPublishMillis = currentMillis;

  if (!mqtt.connected()) return;
  char buf[32];
  itoa(waterAvgADC, buf, 10);          mqtt.publish(TOPIC_WATER_ADC, buf);
  dtostrf(waterVoltage, 1, 3, buf);    mqtt.publish(TOPIC_WATER_VOLT, buf);
  itoa(waterLevelPercent, buf, 10);    mqtt.publish(TOPIC_WATER_LEVEL, buf);
}

/* ============================================================================
   SERIAL MONITOR
   ============================================================================ */
void printSerial(unsigned long currentMillis) {
  if (currentMillis - prevSerialMillis < SERIAL_INTERVAL) return;
  prevSerialMillis = currentMillis;

  DateTime now = getSafeTime();
  char b1[6], b2[6], b3[6], b5[6];
  minToStr(lampOnMin, b1);   minToStr(lampOffMin, b2);
  minToStr(conv1OnMin, b3);  minToStr(conv2OnMin, b5);

  Serial.println(F("\n========================================"));
  Serial.print(F("Sumber waktu : "));
  Serial.println(timeFromNTP ? F("NTP (internet)") : F("RTC (cadangan)"));
  Serial.print(F("Jam          : "));
  if (now.hour() < 10) Serial.print(F("0")); Serial.print(now.hour()); Serial.print(F(":"));
  if (now.minute() < 10) Serial.print(F("0")); Serial.print(now.minute()); Serial.print(F(":"));
  if (now.second() < 10) Serial.print(F("0")); Serial.println(now.second());
  Serial.print(F("Temperature  : ")); Serial.print(temperature); Serial.println(F(" C"));
  Serial.print(F("Humidity     : ")); Serial.print(humidity); Serial.println(F(" %"));
  Serial.print(F("MQ135        : ")); Serial.println(mq135Value);
  Serial.print(F("Water ADC    : ")); Serial.print(waterAvgADC);
  Serial.print(F("  (")); Serial.print(waterLevelPercent); Serial.println(F(" %)"));
  Serial.print(F("Lampu        : ")); Serial.print(lampStatus ? F("ON ") : F("OFF"));
  Serial.print(F("  Mode: ")); Serial.print(lampAutoMode ? F("AUTO") : F("MANUAL"));
  Serial.print(F("  Jadwal: ")); Serial.print(b1); Serial.print(F(" -> ")); Serial.println(b2);
  Serial.print(F("Conveyor     : ")); Serial.print(convPhaseName());
  if (convIsJogging()) Serial.print(F(" [MANUAL]"));
  Serial.print(F("  Jadwal: ")); Serial.print(b3);
  if (conv2Enabled) { Serial.print(F(" & ")); Serial.print(b5); }
  Serial.print(F("  Siklus: maju ")); Serial.print(convRunSec);
  Serial.print(F("s / mundur ")); Serial.print(convRunSec + CONV_REV_EXTRA_MS / 1000.0, 1);
  Serial.print(F("s / jeda ")); Serial.print(convPauseSec);
  Serial.print(F("s / ")); Serial.print(convSpeedPct); Serial.println(F("%"));
  Serial.print(F("Pompa        : ")); Serial.print(pompaStatus ? F("ON ") : F("OFF"));
  Serial.print(F("  Mode: ")); Serial.println(pompaAutoMode ? F("AUTO") : F("MANUAL"));
  Serial.print(F("WiFi         : ")); Serial.print(WiFi.status() == WL_CONNECTED ? F("OK") : F("DISC"));
  Serial.print(F(" (")); Serial.print(WiFi.RSSI()); Serial.println(F(" dBm)"));
  Serial.print(F("MQTT         : ")); Serial.println(mqtt.connected() ? F("OK") : F("DISC"));
  if (!mqtt.connected()) {
    Serial.print(F("MQTT LastErr : ")); Serial.print(mqtt.state());
    Serial.print(F(" = ")); Serial.println(mqttStateString(mqtt.state()));
  }
  Serial.println(F("========================================"));
}

void printWaterDebug(unsigned long currentMillis) {
  if (currentMillis - prevWaterDebugMillis < WATER_DEBUG_INTERVAL) return;
  prevWaterDebugMillis = currentMillis;
  Serial.print(F("[WATER] ADC=")); Serial.print(waterAvgADC);
  Serial.print(F("  V=")); Serial.print(waterVoltage, 3);
  Serial.print(F("  Level=")); Serial.print(waterLevelPercent);
  Serial.print(F("%  Pump=")); Serial.println(pompaStatus ? F("ON") : F("OFF"));
}

/* ============================================================================
   SETUP
   ============================================================================ */
void setup() {
  Serial.begin(115200);
  while (!Serial && millis() < 3000);

  Serial.println(F("\n========================================"));
  Serial.println(F("   SMART CHICKEN COOP v4 - TIMER CONFIG"));
  Serial.println(F("========================================"));

  pinMode(RELAY_LAMP, OUTPUT);
  pinMode(RELAY_POMPA, OUTPUT);
  digitalWrite(RELAY_LAMP, HIGH);
  digitalWrite(RELAY_POMPA, HIGH);
  Serial.println(F("[INIT] Relay OK (Lampu, Pompa)"));

  // Muat jadwal dari NVS sebelum apa pun berjalan
  loadTimerConfig();

  // Motor diinisialisasi SETELAH loadTimerConfig supaya convSpeedPct sudah terisi
  motorInit();

  dht.begin();
  Serial.println(F("[INIT] DHT22 OK"));

  Wire.begin(LCD_SDA, LCD_SCL);
  Wire.setClock(I2C_CLOCK_LCD);
  delay(50);
  lcdAvailable = initLCD();
  if (lcdAvailable) {
    lcdWriteLine(0, " SMART CHICKEN COOP");
    lcdWriteLine(1, "   BENGPUSKOMLEK");
    lcdWriteLine(2, "         X");
    lcdWriteLine(3, "  KADET UNHAN  RI");
  }

  Wire1.begin(RTC_SDA, RTC_SCL);
  Wire1.setClock(100000);
  if (!rtc.begin(&Wire1)) {
    Serial.println(F("[ERROR] RTC NOT FOUND"));
    rtcAvailable = false;
    rtcLastValid = DateTime(F(__DATE__), F(__TIME__));
    //if (lcdAvailable) lcdWriteLine(3, "RTC NOT FOUND!");
  } else {
    Serial.println(F("[INIT] RTC OK"));
    if (rtc.lostPower()) rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    rtcAvailable = true;
    rtcLastValid = rtc.now();
    //if (lcdAvailable) lcdWriteLine(3, "RTC OK");
  }

  analogSetAttenuation(ADC_11db);
  analogSetWidth(12);
  pinMode(MQ135_PIN, INPUT);
  pinMode(WATER_PIN, INPUT);
  Serial.println(F("[INIT] MQ135 & Water Sensor OK"));

  // ===== WIFI =====
  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, pass);
  Serial.print(F("[WIFI] Connecting to ")); Serial.print(ssid);
  //if (lcdAvailable) lcdWriteLine(3, "WiFi connecting...");
  int wifiRetries = 0;
  while (WiFi.status() != WL_CONNECTED && wifiRetries < 40) {
    delay(500); Serial.print(".");
    wifiRetries++;
  }
  Serial.println();
  if (WiFi.status() == WL_CONNECTED) {
    Serial.print(F("[WIFI] CONNECTED! IP: ")); Serial.println(WiFi.localIP());
    Serial.print(F("[WIFI] RSSI: ")); Serial.print(WiFi.RSSI()); Serial.println(F(" dBm"));
    //if (lcdAvailable) lcdWriteLine(3, "WiFi OK");
  } else {
    Serial.println(F("[WIFI] FAILED TO CONNECT!"));
    Serial.print(F("[WIFI] Status: ")); Serial.println(wifiStateString(WiFi.status()));
    //if (lcdAvailable) lcdWriteLine(3, "WiFi GAGAL");
  }

  // ===== NTP =====
  // configTime dipanggil TANPA SYARAT. SNTP akan terus mencoba sendiri
  // begitu WiFi tersambung, walau saat boot jaringannya belum ada.
  configTime(7 * 3600, 0, "pool.ntp.org", "time.google.com");
  if (WiFi.status() == WL_CONNECTED) syncNTP();

  // ===== MQTT =====
  espClient.setInsecure();
  mqtt.setServer(mqtt_server, mqtt_port);
  mqtt.setCallback(mqttCallback);
  mqtt.setBufferSize(512);

  if (WiFi.status() == WL_CONNECTED) {
    reconnectMQTT();
  } else {
    Serial.println(F("[MQTT] SKIP: WiFi not ready"));
  }

  // ===== Isi buffer water sensor =====
  Serial.println(F("[INIT] Pre-filling water buffer..."));
  waterSampleSum = 0;
  for (int i = 0; i < WATER_SAMPLE_COUNT; i++) {
    waterSamples[i] = analogRead(WATER_PIN);
    waterSampleSum += waterSamples[i];
    delay(50);
  }
  waterAvgADC = waterSampleSum / WATER_SAMPLE_COUNT;
  calculateWaterLevel();
  Serial.print(F("[INIT] Water ADC init: ")); Serial.println(waterAvgADC);

  if (lcdAvailable) { delay(1500); lcd->clear(); delay(50); lcdForceRedraw(); }

  Serial.println(F("========================================"));
  Serial.println(F("   Setup Complete!"));
  Serial.println(F("========================================"));
}

/* ============================================================================
   LOOP
   ============================================================================ */
void loop() {
  unsigned long currentMillis = millis();

  if (!mqtt.connected()) {
    if (currentMillis - lastMqttReconnectAttempt >= MQTT_RECONNECT_INTERVAL) {
      lastMqttReconnectAttempt = currentMillis;
      if (reconnectMQTT()) lastMqttReconnectAttempt = 0;
    }
  } else {
    mqtt.loop();
  }

  // Jadwal baru dari dashboard: simpan ke NVS lalu kirim balik konfirmasinya
  if (timerConfigDirty && mqtt.connected()) {
    timerConfigDirty = false;
    saveTimerConfig();
    publishTimerConfig();
  }

  readSensor(currentMillis);
  readWaterSensor(currentMillis);
  calculateWaterLevel();
  wifiTask(currentMillis);
  rtcSyncTask(currentMillis);
  lampAutomation(currentMillis);
  conveyorSchedule();
  conveyorStateMachine();
  pumpAutomation(currentMillis);
  publishMQTT(currentMillis);
  publishWaterData(currentMillis);
  updateLCD(currentMillis);
  lcdHealthCheck(currentMillis);
  printSerial(currentMillis);
  printWaterDebug(currentMillis);
}
