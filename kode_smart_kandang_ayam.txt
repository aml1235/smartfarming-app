bool exhaustStatus = false;

bool pompaStatus = false;

bool lcdAvailable = false;
bool rtcAvailable = false;

unsigned long prevSensorMillis = 0;
const unsigned long SENSOR_INTERVAL = 5000;
unsigned long lastMqttReconnectAttempt = 0;
const unsigned long MQTT_RECONNECT_INTERVAL = 5000;
unsigned long prevPublishMillis = 0;
const unsigned long PUBLISH_INTERVAL = 5000;
unsigned long prevSerialMillis = 0;
const unsigned long SERIAL_INTERVAL = 2000;
unsigned long prevWifiMillis = 0;
const unsigned long WIFI_INTERVAL = 10000;
unsigned long prevLcdMillis = 0;
const unsigned long LCD_INTERVAL = 1000;

DateTime rtcLastValid;
const char* NAMA_HARI[] = {"Minggu","Senin","Selasa","Rabu","Kamis","Jumat","Sabtu"};


/* ============================================================================
   DEBUG MQTT STATE
   ============================================================================ */
const char* mqttStateString(int state) {
  switch (state) {
    case -4: return "TIMEOUT";
    case -3: return "CONN_LOST";
    case -2: return "CONN_FAILED";
    case -1: return "DISCONNECTED";
    case 0:  return "CONNECTED";
    case 1:  return "BAD_PROTOCOL";
    case 2:  return "BAD_CLIENT_ID";
    case 3:  return "UNAVAILABLE";
    case 4:  return "BAD_CREDENTIALS";
    case 5:  return "UNAUTHORIZED";
    default: return "UNKNOWN";
  }
}

/* ============================================================================
   RTC
   ============================================================================ */
bool isRTCValid(DateTime dt) {
  if (dt.year() < 2024 || dt.year() > 2035) return false;
  if (dt.month() < 1 || dt.month() > 12) return false;
  if (dt.day() < 1 || dt.day() > 31) return false;
  if (dt.hour() > 23) return false;
  if (dt.minute() > 59) return false;
  if (dt.second() > 59) return false;
  return true;
}

DateTime getSafeTime() {
  if (!rtcAvailable) return rtcLastValid + TimeSpan(millis()/1000);
  DateTime now = rtc.now();
  if (!isRTCValid(now)) {
    Serial.println(F("[RTC] Invalid! Reset to compile time"));
    rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    now = rtc.now();
  }
  rtcLastValid = now;
  return now;
}

/* ============================================================================
   LCD
   ============================================================================ */
uint8_t scanI2CAddress() {
  const uint8_t addresses[] = {0x27, 0x3F};
  for (uint8_t i = 0; i < 2; i++) {
    Wire.beginTransmission(addresses[i]);
    if (Wire.endTransmission() == 0) return addresses[i];
  }
  for (uint8_t addr = 8; addr < 120; addr++) {
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) return addr;
  }
  return 0;
}

bool initLCD() {
  lcdAddress = scanI2CAddress();
  if (lcdAddress == 0) {
    Serial.println(F("[LCD] NOT DETECTED"));
    return false;
  }
  lcd = new LiquidCrystal_I2C(lcdAddress, LCD_COLS, LCD_ROWS);
  lcd->init(); lcd->backlight();
  lcd->clear();
  lcd->setCursor(0,0); lcd->print("  SMART CHICKEN COOP  ");
  lcd->setCursor(0,1); lcd->print("    MQTT Final Ver    ");
  lcd->setCursor(0,2); lcd->print("   ESP32 DevKit V1    ");
  lcd->setCursor(0,3); lcd->print("   LCD OK! Addr:0x"); lcd->print(lcdAddress, HEX);
  Serial.print(F("[LCD] OK at 0x")); Serial.println(lcdAddress, HEX);
  return true;
}

/* ============================================================================
   MQTT CALLBACK (Perintah dari Laravel)
   ============================================================================ */
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (unsigned int i = 0; i < length; i++) message += (char)payload[i];
  message.trim();
  Serial.print(F("[MQTT] RX [")); Serial.print(topic); Serial.print(F("]: ")); Serial.println(message);

  // Parse JSON command dari Laravel
  StaticJsonDocument<256> doc;
  if (!deserializeJson(doc, message)) {
    if (doc.containsKey("lamp")) {
      lampManualState = (doc["lamp"] == "ON");
      if (!lampAutoMode) { lampStatus = lampManualState; digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH); }
    }
    if (doc.containsKey("motor")) {
      motorStatus = (doc["motor"] == "ON");
      digitalWrite(RELAY_MOTOR, motorStatus ? LOW : HIGH);
    }
    if (doc.containsKey("pompa")) {
      pompaStatus = (doc["pompa"] == "ON");
      digitalWrite(RELAY_POMPA, pompaStatus ? LOW : HIGH);
    }
    if (doc.containsKey("lampAutoMode")) {
      lampAutoMode = (doc["lampAutoMode"] == "ON");
      if (!lampAutoMode) { lampStatus = lampManualState; digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH); }
    }
    
    // Publikasi status kembali ke server secara instan (opsional)
    publishMQTT();
  }
}

/* ============================================================================
   MQTT RECONNECT
   ============================================================================ */
bool reconnectMQTT() {
  Serial.println(F("\n[MQTT] Connecting..."));
  if (mqtt.connect(client_id, mqtt_user, mqtt_pass)) {
    Serial.println(F("[MQTT] CONNECTED!"));
    mqtt.subscribe(TOPIC_CMD);
    Serial.println(F("[MQTT] Subscribed to unified command topic"));
    return true;
  } else {
    int state = mqtt.state();
    Serial.print(F("[MQTT] FAILED: ")); Serial.print(state);
    Serial.print(F(" = ")); Serial.println(mqttStateString(state));
    return false;
  }
}

/* ============================================================================
   SETUP
   ============================================================================ */
void setup() {
  Serial.begin(115200);
  while (!Serial && millis() < 2000);

  Serial.println(F("\n========================================"));
  Serial.println(F("   SMART CHICKEN COOP - MQTT FINAL"));
  Serial.println(F("========================================"));

  pinMode(RELAY_LAMP, OUTPUT); pinMode(RELAY_MOTOR, OUTPUT);
  pinMode(RELAY_POMPA, OUTPUT); pinMode(RELAY_EXHAUST, OUTPUT);
  digitalWrite(RELAY_LAMP, HIGH); digitalWrite(RELAY_MOTOR, HIGH);
  digitalWrite(RELAY_POMPA, HIGH); digitalWrite(RELAY_EXHAUST, HIGH);

  dht.begin();
  Wire.begin(LCD_SDA, LCD_SCL);
  lcdAvailable = initLCD();

  Wire1.begin(RTC_SDA, RTC_SCL);
  if (!rtc.begin(&Wire1)) {
    Serial.println(F("[RTC] NOT FOUND"));
    rtcAvailable = false;
    rtcLastValid = DateTime(F(__DATE__), F(__TIME__));
  } else {
    Serial.println(F("[RTC] OK"));
    if (rtc.lostPower()) rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));
    rtcAvailable = true;
    rtcLastValid = rtc.now();
  }

  analogSetAttenuation(ADC_11db);
  analogSetWidth(12);
  pinMode(MQ135_PIN, INPUT);

  WiFi.mode(WIFI_STA);
  WiFi.begin(ssid, pass);
  Serial.print(F("[WIFI] Connecting"));
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println();
  Serial.print(F("[WIFI] IP: ")); Serial.println(WiFi.localIP());

  espClient.setInsecure();
  mqtt.setServer(mqtt_server, mqtt_port);
  mqtt.setCallback(mqttCallback);
  mqtt.setBufferSize(512);
  reconnectMQTT();

  if (lcdAvailable) { delay(2000); lcd->clear(); }

  Serial.println(F("========================================"));
  Serial.println(F("   Ready! Publishing every 5s"));
  Serial.println(F("========================================"));
}


/* ============================================================================
   FUNGSI PUBLISH MQTT KE LARAVEL
   ============================================================================ */
void publishMQTT() {
    if (mqtt.connected()) {
      StaticJsonDocument<512> doc;
      doc["temperature"] = round(temperature * 10) / 10.0;
      doc["humidity"] = round(humidity * 10) / 10.0;
      doc["ammonia"] = mq135Value; 
      doc["mq135Voltage"] = mq135Voltage;
      
      // Status Relay
      doc["lampStatus"] = lampStatus ? "ON" : "OFF";
      doc["exhaustStatus"] = exhaustStatus ? "ON" : "OFF";
      doc["motorStatus"] = motorStatus ? "ON" : "OFF";
      doc["pumpStatus"] = pompaStatus ? "ON" : "OFF"; 
      doc["lampAutoMode"] = lampAutoMode ? "ON" : "OFF";

      String payload; 
      serializeJson(doc, payload);
      bool ok = mqtt.publish(TOPIC_SENSOR, payload.c_str());
      
      Serial.print(F("[PUB] Sending Data JSON... "));
      Serial.println(ok ? F("OK") : F("FAIL"));
    }
}


/* ============================================================================
   LOOP
   ============================================================================ */
void loop() {
  unsigned long currentMillis = millis();

  // MQTT
  if (!mqtt.connected()) {
    if (currentMillis - lastMqttReconnectAttempt >= MQTT_RECONNECT_INTERVAL) {
      lastMqttReconnectAttempt = currentMillis;
      if (reconnectMQTT()) lastMqttReconnectAttempt = 0;
    }
  } else {
    mqtt.loop();
  }

  // Sensor
  if (currentMillis - prevSensorMillis >= SENSOR_INTERVAL) {
    prevSensorMillis = currentMillis;
    float h = dht.readHumidity();
    float t = dht.readTemperature();
    if (!isnan(h) && !isnan(t)) { temperature = t; humidity = h; }
    mq135Value = analogRead(MQ135_PIN);
    mq135Voltage = mq135Value * (3.3 / 4095.0);
  }

  // Publish MQTT Terpusat
  if (currentMillis - prevPublishMillis >= PUBLISH_INTERVAL) {
    prevPublishMillis = currentMillis;
    publishMQTT();
  }

  // WiFi reconnect
  if (currentMillis - prevWifiMillis >= WIFI_INTERVAL) {
    prevWifiMillis = currentMillis;
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println(F("[WIFI] Reconnecting..."));
      WiFi.disconnect(); WiFi.begin(ssid, pass);
    }
  }

  // Lamp automation
  static bool lastAutoMode = true;
  if (lampAutoMode) {
    if (!lastAutoMode) lastAutoMode = true;
    DateTime now = getSafeTime();
    bool shouldLampOn = (now.hour() >= 18 || now.hour() < 6);
    if (shouldLampOn != lampStatus) {
      lampStatus = shouldLampOn;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      publishMQTT();
    }
  } else {
    if (lastAutoMode) {
      lastAutoMode = false;
      lampStatus = lampManualState;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      publishMQTT();
    } else if (lampStatus != lampManualState) {
      lampStatus = lampManualState;
      digitalWrite(RELAY_LAMP, lampStatus ? LOW : HIGH);
      publishMQTT();
    }
  }

  // Exhaust automation
  bool shouldExhaustOn = false;
  if (!exhaustActive) {
    if (temperature >= 32.0 || mq135Value >= 250) shouldExhaustOn = true;
  } else {
    if (temperature < 31.0 && mq135Value < 220) shouldExhaustOn = false;
    else shouldExhaustOn = true;
  }
  if (shouldExhaustOn != exhaustStatus) {
    exhaustStatus = shouldExhaustOn;
    exhaustActive = shouldExhaustOn;
    digitalWrite(RELAY_EXHAUST, exhaustStatus ? LOW : HIGH);
    publishMQTT();
  }

  // LCD
  if (lcdAvailable && currentMillis - prevLcdMillis >= LCD_INTERVAL) {
    prevLcdMillis = currentMillis;
    DateTime now = getSafeTime();
    lcd->setCursor(0,0); lcd->print("Suhu  : "); lcd->print(temperature,1); lcd->print(" C  ");
    lcd->setCursor(0,1); lcd->print("Amonia: "); lcd->print(mq135Value); lcd->print("    ");
    lcd->setCursor(0,2); lcd->print("MQTT  : "); lcd->print(mqtt.connected()?"ONLINE ":"OFFLINE");
    lcd->setCursor(0,3); lcd->print("Time  : ");
    if (now.hour()<10) lcd->print("0"); lcd->print(now.hour()); lcd->print(":");
    if (now.minute()<10) lcd->print("0"); lcd->print(now.minute());
  }

  // Serial
  if (currentMillis - prevSerialMillis >= SERIAL_INTERVAL) {
    prevSerialMillis = currentMillis;
    DateTime now = getSafeTime();
    Serial.println(F("\n========================================"));
    Serial.print(F("Jam         : ")); if (now.hour()<10) Serial.print(F("0")); Serial.print(now.hour());
    Serial.print(F(":")); if (now.minute()<10) Serial.print(F("0")); Serial.print(now.minute());
    Serial.print(F(":")); if (now.second()<10) Serial.print(F("0")); Serial.println(now.second());
    Serial.print(F("Temp        : ")); Serial.print(temperature); Serial.println(F(" C"));
    Serial.print(F("Humidity    : ")); Serial.print(humidity); Serial.println(F(" %"));
    Serial.print(F("MQ135       : ")); Serial.println(mq135Value);
    Serial.print(F("Lampu       : ")); Serial.println(lampStatus?F("ON"):F("OFF"));
    Serial.print(F("Exhaust     : ")); Serial.println(exhaustStatus?F("ON"):F("OFF"));
    Serial.println(F("========================================"));
  }
}
