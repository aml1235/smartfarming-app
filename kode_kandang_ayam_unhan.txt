/*
  ============================================================================
  SMART CHICKEN COOP  —  v6.1
  Expansion board polos · ADC internal · tanpa ADS1115 · tanpa LVD
  ============================================================================

  BARU DI v6.1

    1. URUTAN SIKLUS PAKAN DIPERBAIKI
         Motor maju DULU, servo baru membuka setelah jeda.
         1. Motor maju
         2. Tunggu jeda detik  (dudukan menempuh jarak awal)
         3. Servo buka
         4. Servo tutup: timer "lama buka" habis ATAU limit A tercapai
         5. Motor terus sampai limit A
         6. Jeda 2 detik
         7. Motor mundur sampai limit B

    2. LOGIKA PELAMPUNG TIGA TINGKAT
         bawah=0 atas=0  ->  KOSONG   (pompa menyala)
         bawah=1 atas=0  ->  SEDANG   (pompa lanjut mengisi)
         bawah=1 atas=1  ->  PENUH    (pompa berhenti)
         bawah=0 atas=1  ->  ERROR    (mustahil — pelampung macet)

       Histeresis alami: mengisi dari KOSONG sampai PENUH, lalu diam sampai
       benar-benar KOSONG lagi. Tidak berkedut di posisi SEDANG.

       Kondisi ERROR memblokir pompa dan mengirim alarm. Air tidak mungkin
       menyentuh pelampung atas tanpa melewati yang bawah, jadi pembacaan itu
       berarti ada pelampung tersangkut.

  PIN RELAI
    Lampu = GPIO 21   ·   Pompa = GPIO 47

  PENGAWATAN
    Buck 1 : 12 V -> 5,0 V  masuk lewat PIN 5V + GND header expansion.
    Buck 2 : 12 V -> 6,0 V  ke servo MG996R. Elko 1000 uF di terminal servo.
    Perangkat I2C mengambil VCC dari 3V3, BUKAN 5V (tanpa level shifter).
    Sensor tegangan -> GPIO 1. MQ-135 AO -> pembagi 2:1 -> GPIO 2.
    HC-SR04 ECHO -> pembagi 1k/2k -> GPIO 38.
    BTS7960: RPWM 4, LPWM 5, R_EN+L_EN 6, VCC 3V3, GND perfboard.

  PERINTAH SERIAL
    s status    d diagnostik   i pindai I2C   B pulihkan bus   N sinkron NTP
    c konfig    D status daya  m status MQTT  u uji relai
    w pompa isi / berhenti (toggle)     W pompa berhenti
    r reset kunci pompa
    1 lampu ON  0 lampu OFF    a lampu AUTO
    f siklus pakan   > maju   < mundur   x STOP DARURAT
    [ servo buka     ] servo tutup       k reset kunci motor
    P kalibrasi jarak PENUH   K kalibrasi jarak KOSONG
    Q kalibrasi R0 MQ-135     M hidup/matikan MQ-135
    v12.73 kalibrasi tegangan · S12.15 simulasi tegangan · S hentikan
    T uji Telegram            Z hapus NVS

  PUSTAKA
    Adafruit SHT31 · RTClib · LiquidCrystal I2C · ESP32Servo · PubSubClient
  ============================================================================
*/

void setup() {
  // Aktuator dimatikan lebih dulu, sebelum apa pun
  pinMode(PIN_RELAI_LAMPU,OUTPUT); digitalWrite(PIN_RELAI_LAMPU,RELAI_MATI);
  pinMode(PIN_RELAI_POMPA,OUTPUT); digitalWrite(PIN_RELAI_POMPA,RELAI_MATI);
  pinMode(PIN_MOTOR_EN,OUTPUT);    digitalWrite(PIN_MOTOR_EN,LOW);
  pinMode(PIN_MOTOR_RPWM,OUTPUT);  digitalWrite(PIN_MOTOR_RPWM,LOW);
  pinMode(PIN_MOTOR_LPWM,OUTPUT);  digitalWrite(PIN_MOTOR_LPWM,LOW);
  pinMode(PIN_MQ_HEATER,OUTPUT);   digitalWrite(PIN_MQ_HEATER,LOW);

  pinMode(PIN_PELAMPUNG_ATAS,INPUT_PULLUP);
  pinMode(PIN_PELAMPUNG_BAWAH,INPUT_PULLUP);
  pinMode(PIN_LIMIT_A,INPUT_PULLUP);
  pinMode(PIN_LIMIT_B,INPUT_PULLUP);
  pinMode(PIN_TRIG,OUTPUT); digitalWrite(PIN_TRIG,LOW);
  pinMode(PIN_ECHO,INPUT);

  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);

  Serial.begin(115200); Serial.setTimeout(50); delay(300);
  Serial.println(F("\n\n=== SMART CHICKEN COOP v6.1 ==="));
  Serial.println(F("Relai: LAMPU=GPIO21  POMPA=GPIO47"));

  Wire.begin(PIN_SDA,PIN_SCL,I2C_FREQ); Wire.setTimeOut(100); delay(50);
  pindaiI2C(); siapkanLcd(); siapkanSht31(); siapkanRtc();

  nvs.begin("smartcoop",false);
  muatKonfigurasi();
  muatStatusTersimpan();
  if (mq135Aktif) { digitalWrite(PIN_MQ_HEATER,HIGH); tsMqNyala=millis(); }

  mulaiWifi();
  netTls.setInsecure();
  mqtt.setServer(MQTT_HOST,MQTT_PORT);
  mqtt.setCallback(mqttPesanMasuk);
  mqtt.setBufferSize(1024);
  mqtt.setKeepAlive(30);

  // Baca pelampung apa adanya supaya tidak ada lonjakan saat start
  rawAtas  = (digitalRead(PIN_PELAMPUNG_ATAS)==PELAMPUNG_AKTIF);
  rawBawah = (digitalRead(PIN_PELAMPUNG_BAWAH)==PELAMPUNG_AKTIF);
  pelAtas=rawAtas; pelBawah=rawBawah; hitungStatusTangki();

  bacaLimit();
  if (limitA&&limitB) Serial.println(F("[LIMIT] KEDUA limit aktif — periksa kabel"));
  if (statusPompa==POMPA_KUNCI) Serial.println(F("[POMPA] TERKUNCI dari sesi lalu — ketik 'r'"));
  if (fasePakan==PAKAN_KUNCI)   Serial.println(F("[PAKAN] TERKUNCI dari sesi lalu — ketik 'k'"));

  Serial.printf("\nTangki    : %s (bawah=%d atas=%d)\n",
    namaTangki(), pelBawah, pelAtas);
  Serial.printf("Mode otomatis: pompa=%s lampu=%s pakan=%s\n",
    otoPompa?"ON":"OFF", otoLampu?"ON":"OFF", otoPakan?"ON":"OFF");
  Serial.println(F("*** LVD TIDAK AKTIF · DETEKSI MACET NONAKTIF ***"));
  Serial.printf("Batas waktu motor %u detik — setel ke waktu tempuh + 3.\n\n", batasMotorDtk);

  tsSensor=tsHalaman=tsCekWifi=tsPompaBerhenti=tsTegangan=tsTerbit=millis();
  bacaSensor(); gambarHalaman(); cetakStatus();
}


// ===========================================================================
//  LOOP
// ===========================================================================
void loop() {
  uint32_t kini=millis();

  urusSerial(); bacaPelampung(); bacaLimit();
  urusPompa(); urusLampu(); urusPakan();
  if (mqtt.connected()) mqtt.loop();

  if (kini-tsTegangan>=INTERVAL_TEGANGAN) { tsTegangan=kini; bacaTegangan(); }
  if (kini-tsMq>=INTERVAL_MQ)             { tsMq=kini;       bacaMq135(); }
  if (kini-tsSensor>=INTERVAL_SENSOR)     { tsSensor=kini;   bacaSensor(); }
  if (kini-tsHalaman>=INTERVAL_HALAMAN)   { tsHalaman=kini;  halaman=(halaman+1)%JUMLAH_HALAMAN; gambarHalaman(); }
  if (kini-tsCekWifi>=INTERVAL_CEK_WIFI)  { tsCekWifi=kini;  jagaWifi(); }
  if (kini-tsTerbit>=INTERVAL_TERBIT)     { tsTerbit=kini;   terbitkanStatus(); }

  // Sonar memblokir ~300 ms — hanya saat mekanisme diam
  if (kini-tsUkurPakan>=INTERVAL_UKUR_PAKAN && fasePakan==PAKAN_IDLE
      && arahManual==MANUAL_DIAM) { tsUkurPakan=kini; ukurLevelPakan(); }

  if (WiFi.status()==WL_CONNECTED) {
    if (!mqtt.connected() && kini-tsMqttUlang>=INTERVAL_MQTT_ULANG) {
      tsMqttUlang=kini; sambungMqtt();
    }
    if (!ntpPernahSukses || kini-tsNtp>=INTERVAL_NTP) { tsNtp=kini; sinkronkanNtp(); }
  }
}


// ===========================================================================
//  PELAMPUNG — logika tiga tingkat
// ===========================================================================
/*
  bawah=0 atas=0  ->  KOSONG   air belum menyentuh pelampung bawah
  bawah=1 atas=0  ->  SEDANG   air di antara kedua pelampung
  bawah=1 atas=1  ->  PENUH    air menyentuh pelampung atas
  bawah=0 atas=1  ->  ERROR    mustahil secara fisik, pelampung macet

  Histeresis alami: pompa menyala di KOSONG, berhenti di PENUH, dan diam
  sepanjang SEDANG. Tidak ada kedutan di sekitar satu titik ambang.
*/
void hitungStatusTangki() {
  if      (!pelBawah && !pelAtas) statusTangki = TANGKI_KOSONG;
  else if ( pelBawah && !pelAtas) statusTangki = TANGKI_SEDANG;
  else if ( pelBawah &&  pelAtas) statusTangki = TANGKI_PENUH;
  else                            statusTangki = TANGKI_ERROR;
}

const char* namaTangki() {
  switch (statusTangki) {
    case TANGKI_KOSONG: return "KOSONG";
    case TANGKI_SEDANG: return "SEDANG";
    case TANGKI_PENUH:  return "PENUH";
    default:            return "ERROR";
  }
}

void bacaPelampung() {
  uint32_t kini=millis();

  bool ba=(digitalRead(PIN_PELAMPUNG_ATAS)==PELAMPUNG_AKTIF);
  if (ba!=rawAtas) { rawAtas=ba; tsDebAtas=kini; }
  else if (kini-tsDebAtas>=DEBOUNCE_PELAMPUNG) pelAtas=rawAtas;

  bool bb=(digitalRead(PIN_PELAMPUNG_BAWAH)==PELAMPUNG_AKTIF);
  if (bb!=rawBawah) { rawBawah=bb; tsDebBawah=kini; }
  else if (kini-tsDebBawah>=DEBOUNCE_PELAMPUNG) pelBawah=rawBawah;

  StatusTangki lama = statusTangki;
  hitungStatusTangki();

  if (statusTangki==TANGKI_ERROR && lama!=TANGKI_ERROR) {
    Serial.println(F("[TANGKI] ERROR — pelampung atas aktif tapi bawah tidak."));
    Serial.println(F("[TANGKI] Air tidak mungkin begitu. Periksa pelampung macet."));
    alarm(tsAlarmTangki,"Pelampung tangki bermasalah: atas aktif tapi bawah tidak. "
                        "Pompa diblokir sampai diperbaiki.");
  }
}


// ===========================================================================
//  POMPA
// ===========================================================================
void urusPompa() {
  uint32_t kini=millis();
  switch (statusPompa) {

    case POMPA_SIAP:
      // Menyala hanya saat benar-benar KOSONG. Pemicu otomatis dapat
      // dimatikan dari dashboard; manual tetap bisa.
      if (otoPompa && statusTangki==TANGKI_KOSONG) nyalakanPompa();
      break;

    case POMPA_JALAN:
      /*
        Mode manual: pelampung diabaikan, hanya batas waktu yang berlaku.
        Tidak dihitung sebagai kegagalan, jadi kunci dry-run tidak terpicu
        oleh demonstrasi manual.
      */
      if (pompaManual) {
        if (kini-tsPompaMulai >= POMPA_MANUAL_MAKS) {
          matikanPompa();
          Serial.println(F("[POMPA] Batas manual habis"));
        }
        break;
      }

      if (statusTangki==TANGKI_PENUH) {
        matikanPompa(); gagalPompaBeruntun=0; tsAlarmPompa=0;
        simpanU8("gagalP",NV_GAGAL_POMPA,0);
        siklusPompa++;
        simpanU16("sikPompa",NV_SIKLUS_HI,NV_SIKLUS_LO,siklusPompa);
        if (rtcSehat) {
          isiJam=waktuKini.hour(); isiMenit=waktuKini.minute();
          simpanU8("isiJ",NV_ISI_JAM,isiJam);
          simpanU8("isiM",NV_ISI_MENIT,isiMenit);
        }
        Serial.printf("[POMPA] Selesai, siklus ke-%u\n",siklusPompa);
        terbitkanStatus();
      } else if (kini-tsPompaMulai>=POMPA_BATAS_WAKTU) {
        // Proteksi dry-run — TETAP bekerja meski mode otomatis dimatikan
        matikanPompa(); gagalPompaBeruntun++;
        simpanU8("gagalP",NV_GAGAL_POMPA,gagalPompaBeruntun);
        Serial.printf("[POMPA] BATAS WAKTU — gagal ke-%u\n",gagalPompaBeruntun);
        if (gagalPompaBeruntun>=POMPA_MAKS_GAGAL) {
          statusPompa=POMPA_KUNCI;
          simpanU8("kunciP",NV_KUNCI_POMPA,1);
          Serial.println(F("[POMPA] TERKUNCI"));
          alarm(tsAlarmPompa,"Pompa terkunci setelah 3 kali gagal mengisi. Periksa sumber air.");
        }
      }
      break;

    case POMPA_JEDA:
      if (kini-tsPompaBerhenti>=POMPA_JEDA_SIKLUS) statusPompa=POMPA_SIAP;
      break;

    case POMPA_KUNCI: break;
  }
}

void nyalakanPompa() {
  digitalWrite(PIN_RELAI_POMPA,RELAI_NYALA);
  statusPompa=POMPA_JALAN; tsPompaMulai=millis();
  Serial.println(F("[POMPA] Menyala"));
}

void matikanPompa() {
  digitalWrite(PIN_RELAI_POMPA,RELAI_MATI);
  statusPompa=POMPA_JEDA; tsPompaBerhenti=millis();
  pompaManual=false;
}

// Toggle: menekan saat berjalan akan menghentikan
void isiSekarang() {
  if (statusPompa==POMPA_KUNCI) { Serial.println(F("[POMPA] Terkunci — ketik 'r'")); return; }
  if (statusPompa==POMPA_JALAN) {
    matikanPompa(); Serial.println(F("[POMPA] Dihentikan manual")); return;
  }
  if (statusTangki==TANGKI_ERROR) {
    Serial.println(F("[POMPA] Diblokir — pelampung bermasalah")); return;
  }
  pompaManual = (statusTangki==TANGKI_PENUH);
  if (pompaManual)
    Serial.printf("[POMPA] Tangki PENUH — mode manual, berhenti otomatis %lu detik\n",
                  (unsigned long)(POMPA_MANUAL_MAKS/1000));
  nyalakanPompa();
}

}

// ===========================================================================
//  MOTOR
// ===========================================================================
void motorBerhenti() {
  analogWrite(PIN_MOTOR_RPWM,0); analogWrite(PIN_MOTOR_LPWM,0);
  digitalWrite(PIN_MOTOR_EN,LOW);
}

// Soft-start 300 ms — mencegah lonjakan yang bisa mereset ESP32
void motorJalan(bool maju) {
  motorBerhenti(); delay(30);
  digitalWrite(PIN_MOTOR_EN,HIGH);
  uint8_t pin = maju?PIN_MOTOR_RPWM:PIN_MOTOR_LPWM;
  for (int d=0;d<=255;d+=15) { analogWrite(pin,d); delay(18); }
  analogWrite(pin,255);
  tsMotorMulai=millis();
}

void bacaLimit() {
  limitA=(digitalRead(PIN_LIMIT_A)==LIMIT_TERSENTUH);
  limitB=(digitalRead(PIN_LIMIT_B)==LIMIT_TERSENTUH);
}

void kunciMotor(const char* sebab) {
  motorBerhenti();
  if (servoPakan.attached()) { servoPakan.write(sudutTutup); delay(WAKTU_SERVO); servoPakan.detach(); }
  fasePakan=PAKAN_KUNCI; arahManual=MANUAL_DIAM; sebabKunciMotor=sebab;
  simpanU8("kunciM",NV_KUNCI_MOTOR,1);
  Serial.printf("[PAKAN] TERKUNCI — %s\n",sebab);
  alarm(tsAlarmMotor,"Unit pakan terkunci: "+String(sebab)+". Periksa mekanisme.");
}


// ===========================================================================
//  SIKLUS PAKAN
//  Urutan: motor maju -> jeda -> servo buka -> servo tutup -> limit A ->
//          jeda 2 detik -> motor mundur -> limit B
// ===========================================================================
void mulaiSiklusPakan() {
  if (fasePakan==PAKAN_KUNCI) { Serial.println(F("[PAKAN] Terkunci")); return; }
  if (fasePakan!=PAKAN_IDLE)  { Serial.println(F("[PAKAN] Siklus berjalan")); return; }
  if (limitA&&limitB)         { Serial.println(F("[PAKAN] Kedua limit aktif")); return; }

  arahManual=MANUAL_DIAM;
  servoPakan.attach(PIN_SERVO,500,2500);      // rentang pulsa MG996R
  servoPakan.write(sudutTutup);
  servoSudahBuka=false; servoSudahTutup=false;

  if (limitA) {
    // Sudah di ujung jauh — tidak ada yang perlu ditempuh, langsung mundur
    Serial.println(F("[PAKAN] Sudah di limit A — langsung fase mundur"));
    servoSudahBuka=true; servoSudahTutup=true;
    fasePakan=PAKAN_TAHAN_UJUNG; tsFase=millis();
    return;
  }

  motorJalan(true);
  fasePakan=PAKAN_MAJU; tsFase=millis();
  Serial.printf("[PAKAN] Motor maju — servo buka setelah %u detik\n", jedaAwalDtk);
}

void urusPakan() {
  uint32_t kini=millis();

  // Pemicu jadwal dapat dimatikan dari dashboard. Manual tetap bisa.
  if (otoPakan && fasePakan==PAKAN_IDLE && rtcSehat
      && waktuKini.minute()!=menitTerakhirPakan) {
    for (uint8_t i=0;i<MAKS_JADWAL_PAKAN;i++)
      if (jadwalPakan[i].aktif && jadwalPakan[i].jam==waktuKini.hour()
          && jadwalPakan[i].menit==waktuKini.minute()) {
        menitTerakhirPakan=waktuKini.minute();
        Serial.printf("[PAKAN] Jadwal %02d:%02d memicu siklus\n",jadwalPakan[i].jam,jadwalPakan[i].menit);
        mulaiSiklusPakan(); break;
      }
  }

  if (arahManual!=MANUAL_DIAM) {
    if ((arahManual==MANUAL_MAJU&&limitA)||(arahManual==MANUAL_MUNDUR&&limitB)) {
      motorBerhenti(); arahManual=MANUAL_DIAM; Serial.println(F("[MOTOR] Berhenti di limit"));
    } else if (kini-tsMotorMulai>=(uint32_t)batasMotorDtk*1000UL) {
      motorBerhenti(); arahManual=MANUAL_DIAM; Serial.println(F("[MOTOR] Batas waktu manual"));
    }
  }

  switch (fasePakan) {
    case PAKAN_IDLE: case PAKAN_KUNCI: break;

    case PAKAN_MAJU:
      /*
        Servo BUKA setelah jeda dihitung dari saat motor mulai bergerak.
        Ujung bukaan pakan tidak langsung berada di atas tempat pakan, jadi
        dudukan perlu menempuh jarak awal sebelum pintu dibuka.
      */
      if (!servoSudahBuka && kini-tsMotorMulai >= (uint32_t)jedaAwalDtk*1000UL) {
        servoPakan.write(sudutBuka);
        servoSudahBuka=true; tsServoBuka=kini;
        Serial.printf("[PAKAN] Servo buka %u derajat\n",sudutBuka);
      }

      // Servo TUTUP pada mana pun yang lebih dulu: timer habis ATAU limit A
      if (servoSudahBuka && !servoSudahTutup &&
          (limitA || kini-tsServoBuka >= (uint32_t)lamaBukaDtk*1000UL)) {
        servoPakan.write(sudutTutup); servoSudahTutup=true;
        Serial.println(F("[PAKAN] Servo tutup"));
      }

      if (kini-tsMotorMulai>=(uint32_t)batasMotorDtk*1000UL) {
        kunciMotor("batas waktu maju — limit A tidak tercapai"); break;
      }
      if (limitA) {
        motorBerhenti();
        if (!servoSudahTutup) { servoPakan.write(sudutTutup); servoSudahTutup=true; }
        fasePakan=PAKAN_TAHAN_UJUNG; tsFase=kini;
        Serial.println(F("[PAKAN] Limit A — jeda 2 detik"));
      }
      break;

    case PAKAN_TAHAN_UJUNG:
      if (!servoSudahTutup) { servoPakan.write(sudutTutup); servoSudahTutup=true; }
      if (kini-tsFase>=JEDA_UJUNG) {
        if (limitB) { selesaikanSiklus(); break; }
        motorJalan(false); fasePakan=PAKAN_MUNDUR; tsFase=kini;
        Serial.println(F("[PAKAN] Motor mundur"));
      }
      break;

    case PAKAN_MUNDUR:
      if (kini-tsMotorMulai>=(uint32_t)batasMotorDtk*1000UL) {
        kunciMotor("batas waktu mundur — limit B tidak tercapai"); break;
      }
      if (limitB) { motorBerhenti(); selesaikanSiklus(); }
      break;
  }
}

void selesaikanSiklus() {
  motorBerhenti();
  servoPakan.write(sudutTutup); delay(WAKTU_SERVO);
  servoPakan.detach();                 // hemat daya, hilangkan jitter MG996R
  fasePakan=PAKAN_IDLE; siklusPakan++;
  servoSudahBuka=false; servoSudahTutup=false;
  if (rtcSehat) {
    pakanJam=waktuKini.hour(); pakanMenit=waktuKini.minute();
    simpanU8("pkJ",NV_PAKAN_JAM,pakanJam);
    simpanU8("pkM",NV_PAKAN_MENIT,pakanMenit);
  }
  simpanU16("sikPakan",NV_PAKAN_HI,NV_PAKAN_LO,siklusPakan);
  Serial.printf("[PAKAN] Siklus selesai — total %u\n",siklusPakan);
  tsUkurPakan=0;
  terbitkanStatus();
}

void resetKunciMotor() {
  fasePakan=PAKAN_IDLE; sebabKunciMotor=""; tsAlarmMotor=0;
  servoSudahBuka=false; servoSudahTutup=false;
  simpanU8("kunciM",NV_KUNCI_MOTOR,0);
  Serial.println(F("[PAKAN] Kunci direset"));
}

void stopDarurat() {
  motorBerhenti(); arahManual=MANUAL_DIAM;
  if (fasePakan!=PAKAN_KUNCI) fasePakan=PAKAN_IDLE;
  if (servoPakan.attached()) { servoPakan.write(sudutTutup); delay(WAKTU_SERVO); servoPakan.detach(); }
  servoSudahBuka=false; servoSudahTutup=false;
  digitalWrite(PIN_RELAI_POMPA,RELAI_MATI);
  digitalWrite(PIN_RELAI_LAMPU,RELAI_MATI);
  statusPompa=POMPA_JEDA; tsPompaBerhenti=millis(); pompaManual=false;
  lampuNyala=false; modeLampu=LAMPU_MANUAL_OFF; tsManualLampu=millis();
  Serial.println(F("\n*** STOP DARURAT ***\n"));
  terbitkanStatus();
}


// ===========================================================================
//  LAMPU
// ===========================================================================
bool dalamPeriode(uint16_t m, const Periode& p) {
  if (!p.aktif) return false;
  uint16_t a=p.jamOn*60+p.menitOn, b=p.jamOff*60+p.menitOff;
  if (a==b) return false;
  if (a<b) return (m>=a&&m<b);
  return (m>=a||m<b);                       // melewati tengah malam
}

void urusLampu() {
  bool minta=false;

  if (otoLampu && (rtcSehat || ntpPernahSukses)) {
    uint16_t m=waktuKini.hour()*60+waktuKini.minute();
    minta = dalamPeriode(m,lampu1)||dalamPeriode(m,lampu2);

    if (minta!=jadwalSebelumnya) {
      jadwalSebelumnya=minta;
      if (modeLampu!=LAMPU_AUTO) { modeLampu=LAMPU_AUTO;
        Serial.println(F("[LAMPU] Transisi jadwal — kembali AUTO")); }
    }
    if (modeLampu!=LAMPU_AUTO && millis()-tsManualLampu>=LAMPU_BATAS_MANUAL) {
      modeLampu=LAMPU_AUTO; Serial.println(F("[LAMPU] Batas 2 jam habis — AUTO"));
    }
  }

  bool target;
  if (!otoLampu) target = (modeLampu==LAMPU_MANUAL_ON);   // hanya manual
  else target = (modeLampu==LAMPU_MANUAL_ON)?true:(modeLampu==LAMPU_MANUAL_OFF)?false:minta;

  if (target!=lampuNyala) {
    lampuNyala=target;
    digitalWrite(PIN_RELAI_LAMPU,lampuNyala?RELAI_NYALA:RELAI_MATI);
    Serial.printf("[LAMPU] %s\n",lampuNyala?"NYALA":"MATI");
  }
}

void setLampuManual(bool nyala) {
  modeLampu=nyala?LAMPU_MANUAL_ON:LAMPU_MANUAL_OFF;
  tsManualLampu=millis();
  Serial.printf("[LAMPU] Manual %s%s\n", nyala?"NYALA":"MATI",
    otoLampu?" (batas 2 jam)":"");
}


// ===========================================================================
//  ADC INTERNAL — oversampling
// ===========================================================================
float bacaAdcVolt(uint8_t pin) {
  uint32_t total=0;
  for (uint8_t i=0;i<SAMPEL_ADC;i++) { total+=analogReadMilliVolts(pin); delayMicroseconds(120); }
  return (float)total/SAMPEL_ADC/1000.0;
}

void bacaTegangan() {
  if (!isnan(simulasiTegangan)) teganganKini = simulasiTegangan;
  else teganganKini = bacaAdcVolt(PIN_SENSOR_V) * RASIO_PEMBAGI * kalTegangan;

  bufTegangan[idxTegangan]=teganganKini;
  idxTegangan=(idxTegangan+1)%SAMPEL_TEGANGAN;
  if (isiTegangan<SAMPEL_TEGANGAN) isiTegangan++;

  // Perataan 10 detik — motor menjatuhkan tegangan sesaat saat start
  float total=0;
  for (uint8_t i=0;i<isiTegangan;i++) total+=bufTegangan[i];
  teganganRata=total/isiTegangan;

  if (PERINGATAN_AKI && isiTegangan>=SAMPEL_TEGANGAN && teganganRata<AMBANG_PERINGATAN)
    alarm(tsAlarmAki, "Tegangan aki rendah: " + String(teganganRata,2) +
                      " V. Sistem TIDAK melepas beban otomatis — periksa segera.");
}

// Kurva aki timbal cukup linear, berbeda dari LiFePO4 yang hampir datar
int8_t estimasiSoc(float v) {
  if (isnan(v)) return -1;
  if (v>=12.70) return 100;
  if (v>=12.40) return (int8_t)(75+(v-12.40)/0.30*25);
  if (v>=12.20) return (int8_t)(50+(v-12.20)/0.20*25);
  if (v>=12.00) return (int8_t)(25+(v-12.00)/0.20*25);
  if (v>=11.80) return (int8_t)((v-11.80)/0.20*25);
  return 0;
}

const char* statusAki() {
  if (isnan(teganganRata)) return "NORMAL";
  return (teganganRata < AMBANG_PERINGATAN) ? "WASPADA" : "NORMAL";
}


// ===========================================================================
//  MQ-135
// ===========================================================================
void bacaMq135() {
  if (!mq135Aktif) { nh3Ppm=NAN; return; }
  if (millis()-tsMqNyala < MQ_PEMANASAN) { nh3Ppm=NAN; return; }

  float v = bacaAdcVolt(PIN_MQ135) * MQ_PEMBAGI;
  if (v<=0.02 || v>=MQ_VCC-0.02) { nh3Ppm=NAN; return; }

  // Rangkaian MQ: VCC - RS - node - RL - GND, keluaran diukur di node
  mqRs = MQ_RL*(MQ_VCC/v - 1.0);
  if (mq135R0<=0.01) { nh3Ppm=NAN; return; }

  nh3Ppm = 102.2*pow(mqRs/mq135R0, -2.473);
  if (nh3Ppm<0 || nh3Ppm>1000) nh3Ppm=NAN;

  if (!isnan(nh3Ppm) && nh3Ppm>ambangNh3)
    alarm(tsAlarmNh3,"Amonia tinggi: "+String(nh3Ppm,1)+" ppm. Periksa ventilasi kandang.");
}

void kalibrasiMq135() {
  if (!mq135Aktif) { Serial.println(F("[MQ]    Nyalakan dulu dengan 'M'")); return; }
  if (millis()-tsMqNyala<MQ_PEMANASAN) {
    Serial.printf("[MQ]    Masih memanaskan, tunggu %lu detik\n",
      (unsigned long)((MQ_PEMANASAN-(millis()-tsMqNyala))/1000));
    return;
  }
  float total=0; uint8_t n=0;
  for (uint8_t i=0;i<10;i++) {
    float v=bacaAdcVolt(PIN_MQ135)*MQ_PEMBAGI;
    if (v>0.02&&v<MQ_VCC-0.02) { total+=MQ_RL*(MQ_VCC/v-1.0); n++; }
    delay(200);
  }
  if (n<5) { Serial.println(F("[MQ]    Kalibrasi gagal — pembacaan tidak stabil")); return; }
  mq135R0=(total/n)/MQ_RASIO_BERSIH;    // datasheet: RS/R0 udara bersih ~3,6
  nvs.putFloat("mqR0",mq135R0);
  Serial.printf("[MQ]    R0 = %.2f kOhm disimpan\n",mq135R0);
  Serial.println(F("[MQ]    Pastikan tadi sensor di UDARA BERSIH."));
}


// ===========================================================================
//  PENYIMPANAN — DS1307 NVRAM bila ada, NVS bila tidak
// ===========================================================================
void simpanU8(const char* kunci, uint8_t alamat, uint8_t nilai) {
  if (adaRtc) tulisNvramAman(alamat,nilai);
  else        nvs.putUChar(kunci,nilai);
}

uint8_t muatU8(const char* kunci, uint8_t alamat, uint8_t bawaan) {
  uint8_t v;
  if (adaRtc && bacaNvramAman(alamat,v)) return v;
  return nvs.getUChar(kunci,bawaan);
}

void simpanU16(const char* kunci, uint8_t aHi, uint8_t aLo, uint16_t nilai) {
  if (adaRtc) { tulisNvramAman(aHi,nilai>>8); tulisNvramAman(aLo,nilai&0xFF); }
  else        nvs.putUShort(kunci,nilai);
}

uint16_t muatU16(const char* kunci, uint8_t aHi, uint8_t aLo, uint16_t bawaan) {
  uint8_t hi,lo;
  if (adaRtc && bacaNvramAman(aHi,hi) && bacaNvramAman(aLo,lo))
    return ((uint16_t)hi<<8)|lo;
  return nvs.getUShort(kunci,bawaan);
}

void muatStatusTersimpan() {
  siklusPompa        = muatU16("sikPompa",NV_SIKLUS_HI,NV_SIKLUS_LO,0);
  siklusPakan        = muatU16("sikPakan",NV_PAKAN_HI,NV_PAKAN_LO,0);
  gagalPompaBeruntun = muatU8 ("gagalP",NV_GAGAL_POMPA,0);
  isiJam             = muatU8 ("isiJ",NV_ISI_JAM,99);
  isiMenit           = muatU8 ("isiM",NV_ISI_MENIT,0);
  pakanJam           = muatU8 ("pkJ",NV_PAKAN_JAM,99);
  pakanMenit         = muatU8 ("pkM",NV_PAKAN_MENIT,0);

  // Kunci sengaja bertahan melewati restart
  if (muatU8("kunciP",NV_KUNCI_POMPA,0)==1) statusPompa=POMPA_KUNCI;
  if (muatU8("kunciM",NV_KUNCI_MOTOR,0)==1) {
    fasePakan=PAKAN_KUNCI; sebabKunciMotor="dari sesi lalu";
  }
  jumlahBoot = muatU16("boot",NV_BOOT_HI,NV_BOOT_LO,0)+1;
  simpanU16("boot",NV_BOOT_HI,NV_BOOT_LO,jumlahBoot);
}


// ===========================================================================
//  HC-SR04
// ===========================================================================
float pingSekali() {
  digitalWrite(PIN_TRIG,LOW);  delayMicroseconds(4);
  digitalWrite(PIN_TRIG,HIGH); delayMicroseconds(10);
  digitalWrite(PIN_TRIG,LOW);
  uint32_t d=pulseIn(PIN_ECHO,HIGH,TIMEOUT_ECHO);
  if (d==0) return NAN;
  float t=isnan(suhu)?27.0:suhu;
  float v=331.3+0.606*t;                     // kompensasi suhu, m/s
  return (float)d*v/20000.0;                 // cm
}

void ukurLevelPakan() {
  float s[5]; uint8_t n=0;
  for (uint8_t i=0;i<5;i++) {
    float d=pingSekali();
    if (!isnan(d)&&d>=JARAK_MIN_SAH&&d<=JARAK_MAKS_SAH) s[n++]=d;
    delay(JEDA_ANTAR_PING);
  }
  if (n<3) { gagalSonar++; jarakPakan=NAN; persenPakan=-1; return; }

  for (uint8_t i=0;i+1<n;i++) for (uint8_t j=i+1;j<n;j++)
    if (s[j]<s[i]) { float t=s[i]; s[i]=s[j]; s[j]=t; }

  jarakPakan=(n>=5)?(s[1]+s[2]+s[3])/3.0:s[n/2];   // buang ekstrem

  if (jarakKosong>jarakPenuh) {
    float p=(jarakKosong-jarakPakan)/(jarakKosong-jarakPenuh)*100.0;
    persenPakan=(int8_t)constrain(p,0.0f,100.0f);
  } else persenPakan=-1;
}

void kalibrasiJarak(bool penuh) {
  ukurLevelPakan();
  if (isnan(jarakPakan)) { Serial.println(F("[PAKAN] Pengukuran gagal")); return; }
  if (penuh) { jarakPenuh=jarakPakan;  nvs.putFloat("jPenuh",jarakPenuh); }
  else       { jarakKosong=jarakPakan; nvs.putFloat("jKosong",jarakKosong); }
  Serial.printf("[PAKAN] Jarak %s = %.1f cm disimpan\n",penuh?"PENUH":"KOSONG",jarakPakan);
  if (jarakKosong<=jarakPenuh)
    Serial.println(F("[PAKAN] PERINGATAN: jarak kosong harus LEBIH BESAR dari penuh"));
  ukurLevelPakan();
}

void uraiJam(const String& s, uint8_t& jam, uint8_t& menit) {
  int t=s.indexOf(':');
  if (t<1) return;
  jam=constrain(s.substring(0,t).toInt(),0,23);
  menit=constrain(s.substring(t+1).toInt(),0,59);
}

void mqttPesanMasuk(char* topik, byte* data, unsigned int panjang) {
  String t=String(topik), p; p.reserve(panjang);
  for (unsigned int i=0;i<panjang;i++) p+=(char)data[i];
  p.trim();
  Serial.printf("[MQTT]  %s = %s\n",topik,p.c_str());

  if (t=="smartcoop/control/stop")  { stopDarurat(); return; }

  if (t=="smartcoop/control/pompa") {
    if      (p=="isi")  isiSekarang();
    else if (p=="stop") hentikanPompa();
    return;
  }
  if (t=="smartcoop/control/lampu") {
    if (p=="on") setLampuManual(true);
    else if (p=="off") setLampuManual(false);
    else if (p=="auto") { modeLampu=LAMPU_AUTO; Serial.println(F("[LAMPU] AUTO")); }
    return;
  }
  if (t=="smartcoop/control/pakan") { if (p=="mulai") mulaiSiklusPakan(); return; }
  if (t=="smartcoop/control/reset") {
    if (p=="pompa") resetKunciPompa(); else if (p=="motor") resetKunciMotor();
    return;
  }
  if (t=="smartcoop/control/servo") {
    if (fasePakan!=PAKAN_IDLE) return;
    servoPakan.attach(PIN_SERVO,500,2500);
    if (p=="buka") servoPakan.write(sudutBuka);
    else { servoPakan.write(sudutTutup); delay(WAKTU_SERVO); servoPakan.detach(); }
    return;
  }
  if (t=="smartcoop/control/motor") {
    if (p=="stop") { motorBerhenti(); arahManual=MANUAL_DIAM; return; }
    if (fasePakan!=PAKAN_IDLE) return;
    if (p=="maju"   && !limitA) { arahManual=MANUAL_MAJU;   motorJalan(true); }
    if (p=="mundur" && !limitB) { arahManual=MANUAL_MUNDUR; motorJalan(false); }
    return;
  }

  if (!t.startsWith("smartcoop/config/")) return;
  String k=t.substring(17);

  if      (k=="oto_pompa") { otoPompa=(p=="1"); nvs.putBool("otoP",otoPompa);
                             Serial.printf("[MODE]  Pompa otomatis %s\n",otoPompa?"ON":"OFF"); }

String urlEncode(const String& s) {
  String h="0123456789ABCDEF", o="";
  for (uint16_t i=0;i<s.length();i++) {
    char c=s.charAt(i);
    if (isalnum(c)) o+=c;
    else if (c==' ') o+='+';
    else { o+='%'; o+=h[(c>>4)&0xF]; o+=h[c&0xF]; }
  }
  return o;
}

void kirimTelegram(const String& pesan) {
  if (WiFi.status()!=WL_CONNECTED) return;
  if (strlen(TG_TOKEN)<10 || tgChat.length()<4) return;

  WiFiClientSecure klien; klien.setInsecure();
  HTTPClient http;
  String url="https://api.telegram.org/bot"+String(TG_TOKEN)+
             "/sendMessage?chat_id="+tgChat+"&text="+urlEncode(pesan);
  http.setTimeout(5000);
  int kode=-1;
  if (http.begin(klien,url)) { kode=http.GET(); http.end(); }

  Serial.printf("[TG]    (%d) %s\n",kode,pesan.c_str());
  if (kode!=200)
    Serial.println(F("[TG]    GAGAL — periksa chat ID. Dapatkan dari @userinfobot."));
}

void alarm(uint32_t& penanda, const String& pesan) {
  uint32_t kini=millis();
  if (penanda!=0 && kini-penanda<JEDA_ALARM) return;
  penanda=kini;
  kirimTelegram(pesan);
}

void pulihkanI2C() {
  Serial.println(F("[I2C]   Memulihkan bus..."));
  pemulihanBus++;
  Wire.end();
  pinMode(PIN_SCL,OUTPUT); pinMode(PIN_SDA,INPUT_PULLUP);
  for (uint8_t i=0;i<9;i++) {              // bebaskan slave yang menahan SDA
    digitalWrite(PIN_SCL,HIGH); delayMicroseconds(10);
    digitalWrite(PIN_SCL,LOW);  delayMicroseconds(10);
  }
  digitalWrite(PIN_SCL,HIGH); delayMicroseconds(10);
  pinMode(PIN_SDA,OUTPUT);
  digitalWrite(PIN_SDA,LOW);  delayMicroseconds(10);
  digitalWrite(PIN_SDA,HIGH); delayMicroseconds(10);
  Wire.begin(PIN_SDA,PIN_SCL,I2C_FREQ); Wire.setTimeOut(100); delay(20);
}

void pindaiI2C() {
  Serial.println(F("\n--- Memindai bus I2C ---"));
  uint8_t n=0;
  for (uint8_t a=1;a<127;a++) {
    Wire.beginTransmission(a);
    if (Wire.endTransmission()==0) {
      n++; Serial.printf("  0x%02X  ",a);
      if (a==ALAMAT_SHT31_A||a==ALAMAT_SHT31_B){adaSht31=true;alamatSht31=a;Serial.println(F("SHT31"));}
      else if (a==ALAMAT_LCD_A||a==ALAMAT_LCD_B){adaLcd=true;alamatLcd=a;Serial.println(F("LCD 20x4"));}
      else if (a==ALAMAT_EEPROM){Serial.println(F("EEPROM AT24C32 (modul RTC)"));}
      else if (a==ALAMAT_RTC){adaRtc=true;Serial.println(F("DS1307"));}
      else Serial.println(F("tidak dikenali"));
    }
  }
  if (n==0) Serial.println(F("  TIDAK ADA PERANGKAT — periksa SDA/SCL, GND, VCC 3V3."));
  if (!adaRtc) {
    Serial.println(F("  DS1307 tidak muncul. Wajar bila disuplai 3,3 V."));
    Serial.println(F("  Sistem tetap jalan memakai waktu NTP dan penyimpanan NVS."));
  }
  Serial.printf("--- Selesai, %u perangkat ---\n\n",n);
}

bool waktuMasukAkal(const DateTime& t) {
  return (t.year()>=2020&&t.year()<=2099&&t.month()>=1&&t.month()<=12&&
          t.day()>=1&&t.day()<=31&&t.hour()<=23&&t.minute()<=59&&t.second()<=59);
}

bool bacaWaktuAman(DateTime& keluar) {
  if (!adaRtc) return false;
  for (uint8_t c=0;c<3;c++) {
    DateTime t=rtc.now();
    if (waktuMasukAkal(t)) { keluar=t; return true; }
    delay(20);
  }
  return false;
}

bool bacaNvramAman(uint8_t alamat, uint8_t& keluar) {
  if (!adaRtc) return false;
  for (uint8_t c=0;c<3;c++) {
    uint8_t a=rtc.readnvram(alamat); delayMicroseconds(300);
    uint8_t b=rtc.readnvram(alamat);
    if (a==b) { keluar=a; return true; }
    delay(10);
  }
  gagalNvram++; return false;
}

bool tulisNvramAman(uint8_t alamat, uint8_t nilai) {
  if (!adaRtc) return false;
  for (uint8_t c=0;c<3;c++) {
    rtc.writenvram(alamat,nilai); delay(5);
    uint8_t cek;
    if (bacaNvramAman(alamat,cek)&&cek==nilai) return true;
  }
  gagalNvram++; return false;
}

void siapkanSht31() {
  if (!adaSht31) { Serial.println(F("[SHT31] tidak terdeteksi")); return; }
  if (!sht31.begin(alamatSht31)) { adaSht31=false; Serial.println(F("[SHT31] gagal init")); return; }
  sht31.heater(false);
  Serial.printf("[SHT31] siap di 0x%02X\n",alamatSht31);
}

void siapkanRtc() {
  if (!adaRtc) { Serial.println(F("[RTC]   tidak ada — memakai NTP dan NVS")); return; }
  if (!rtc.begin()) { adaRtc=false; Serial.println(F("[RTC]   gagal init")); return; }
  if (!rtc.isrunning()) {
    Serial.println(F("[RTC]   osilator berhenti — diisi waktu kompilasi"));
    rtc.adjust(DateTime(F(__DATE__),F(__TIME__))); delay(20);
  }
  DateTime t;
  if (!bacaWaktuAman(t)) {
    // Bus tidak dapat dipercaya — sengaja TIDAK menulis apa pun ke RTC
    rtcSehat=false; gagalRtc++;
    Serial.println(F("[RTC]   PEMBACAAN AWAL RUSAK — tidak ada penulisan"));
    return;
  }
  rtcSehat=true; waktuKini=t;
  uint8_t hi,lo;
  if (bacaNvramAman(NV_DRIFT_HI,hi)&&bacaNvramAman(NV_DRIFT_LO,lo))
    driftTerakhir=(int16_t)(((uint16_t)hi<<8)|lo);
  Serial.printf("[RTC]   siap — %02d/%02d/%04d %02d:%02d\n",
    waktuKini.day(),waktuKini.month(),waktuKini.year(),waktuKini.hour(),waktuKini.minute());
}

void bacaSensor() {
  if (adaSht31) {
    float t=sht31.readTemperature(),h=sht31.readHumidity();
    if (!isnan(t)&&t>-40&&t<125&&!isnan(h)&&h>=0&&h<=100) { suhu=t; kelembapan=h; }
    else gagalSht++;
  }
  if (adaRtc) {
    DateTime t;
    if (bacaWaktuAman(t)) { waktuKini=t; rtcSehat=true; gagalRtcBeruntun=0; }
    else {
      rtcSehat=false; gagalRtc++; gagalRtcBeruntun++;
      if (gagalRtcBeruntun==GAGAL_SEBELUM_PULIH) {
        Serial.println(F("[RTC]   Gagal 5x beruntun — memulihkan bus"));
        pulihkanI2C();
      }
      // Waktu lama DIPERTAHANKAN — tidak menebak, tidak menulis
    }
  }
}

void sinkronkanNtp() {
  configTime(OFFSET_WIB,OFFSET_DST,NTP_SERVER_1,NTP_SERVER_2);
  struct tm tn;
  if (!getLocalTime(&tn,5000)) { Serial.println(F("[NTP]   gagal")); return; }
  ntpPernahSukses=true;
  DateTime dtNtp(tn.tm_year+1900,tn.tm_mon+1,tn.tm_mday,tn.tm_hour,tn.tm_min,tn.tm_sec);

  if (!adaRtc) { waktuKini=dtNtp; return; }

  DateTime dtRtc;
  if (!bacaWaktuAman(dtRtc)) {
    Serial.println(F("[NTP]   RTC tidak terbaca — penulisan DIBATALKAN"));
    waktuKini=dtNtp; return;
  }
  int32_t sel=(int32_t)dtNtp.unixtime()-(int32_t)dtRtc.unixtime();
  int32_t mut=(sel<0)?-sel:sel;
  if (mut>BATAS_KOREKSI_WAJAR) {
    Serial.printf("[NTP]   Koreksi %+ld dtk DITOLAK\n",(long)sel); return;
  }
  if (mut>AMBANG_KOREKSI) {
    rtc.adjust(dtNtp); delay(20);
    DateTime cek;
    if (bacaWaktuAman(cek)) {
      driftTerakhir=(int16_t)constrain(sel,-32000L,32000L);
      tulisNvramAman(NV_DRIFT_HI,(driftTerakhir>>8)&0xFF);
      tulisNvramAman(NV_DRIFT_LO,driftTerakhir&0xFF);
      waktuKini=cek; rtcSehat=true;
      Serial.printf("[NTP]   RTC dikoreksi %+ld detik\n",(long)sel);
    } else { Serial.println(F("[NTP]   Verifikasi gagal")); rtcSehat=false; }
  } else {
    Serial.printf("[NTP]   RTC akurat (%+ld dtk)\n",(long)sel);
    waktuKini=dtRtc; rtcSehat=true;
  }
}

void urusSerial() {
  if (!Serial.available()) return;
  String baris=Serial.readStringUntil('\n'); baris.trim();
  if (baris.length()==0) return;
  char c=baris.charAt(0);
  String arg=baris.substring(1); arg.trim();

  switch (c) {
    case 's': cetakStatus(); break;
    case 'd': cetakDiagnostik(); break;
    case 'i': pindaiI2C(); break;
    case 'B': pulihkanI2C(); break;
    case 'N': sinkronkanNtp(); break;
    case 'c': cetakKonfigurasi(); break;
    case 'D': cetakDaya(); break;
    case 'm': cetakMqtt(); break;
    case 'u': ujiRelai(); break;
    case 'w': isiSekarang(); break;
    case 'W': hentikanPompa(); break;
    case 'r': resetKunciPompa(); break;
    case '1': setLampuManual(true); break;
    case '0': setLampuManual(false); break;
    case 'a': modeLampu=LAMPU_AUTO; Serial.println(F("[LAMPU] AUTO")); break;
    case 'f': mulaiSiklusPakan(); break;
    case 'k': resetKunciMotor(); break;
    case 'x': stopDarurat(); break;
    case 'P': kalibrasiJarak(true); break;
    case 'K': kalibrasiJarak(false); break;
    case 'Q': kalibrasiMq135(); break;
    case 'T': kirimTelegram("Uji koneksi Smart Chicken Coop. Sistem tersambung."); break;
    case 'Z': nvs.clear(); Serial.println(F("[NVS]   Dihapus — restart untuk default")); break;

    case 'M':
      mq135Aktif=!mq135Aktif; nvs.putBool("mqAkt",mq135Aktif);
      digitalWrite(PIN_MQ_HEATER,mq135Aktif?HIGH:LOW);
      if (mq135Aktif) { tsMqNyala=millis(); Serial.println(F("[MQ]    NYALA — pemanasan 3 menit")); }
      else { nh3Ppm=NAN; Serial.println(F("[MQ]    MATI")); }
      break;

    case 'v': {
      float ukur=arg.toFloat();
      if (ukur<5.0||ukur>20.0) { Serial.println(F("[KAL]   Format: v12.73")); break; }
      if (isnan(teganganKini)||teganganKini<1.0) { Serial.println(F("[KAL]   Belum ada pembacaan")); break; }
      kalTegangan=kalTegangan*(ukur/teganganKini);
      nvs.putFloat("kalV",kalTegangan);
      Serial.printf("[KAL]   Faktor tegangan = %.4f\n",kalTegangan);
      break;
    }

    case 'S':
      if (arg.length()==0) { simulasiTegangan=NAN; Serial.println(F("[SIM]   Dihentikan")); }
      else { simulasiTegangan=arg.toFloat();
             Serial.printf("[SIM]   Tegangan disimulasikan %.2f V\n",simulasiTegangan); }
      break;

    case '>':
      if (fasePakan!=PAKAN_IDLE||limitA) break;
      arahManual=MANUAL_MAJU; motorJalan(true); Serial.println(F("[MOTOR] Maju"));
      break;

    case '<':
      if (fasePakan!=PAKAN_IDLE||limitB) break;
      arahManual=MANUAL_MUNDUR; motorJalan(false); Serial.println(F("[MOTOR] Mundur"));
      break;

    case '[':
      if (fasePakan!=PAKAN_IDLE) break;
      servoPakan.attach(PIN_SERVO,500,2500); servoPakan.write(sudutBuka);
      Serial.printf("[SERVO] Buka %u\n",sudutBuka);
      break;

    case ']':
      if (fasePakan!=PAKAN_IDLE) break;
      servoPakan.attach(PIN_SERVO,500,2500); servoPakan.write(sudutTutup);
      delay(WAKTU_SERVO); servoPakan.detach();
      Serial.printf("[SERVO] Tutup %u\n",sudutTutup);
      break;

    default: break;
  }
}

void ujiRelai() {
  Serial.println(F("[UJI] LEPAS BEBAN DULU."));
  Serial.println(F("[UJI] LAMPU di GPIO 21, POMPA di GPIO 47."));
  Serial.println(F("[UJI] Klik harus terjadi saat status NYALA.\n"));
  Serial.println(F("[UJI] Lampu NYALA..."));
  digitalWrite(PIN_RELAI_LAMPU,RELAI_NYALA); delay(1500);
  Serial.println(F("[UJI] Lampu MATI"));
  digitalWrite(PIN_RELAI_LAMPU,RELAI_MATI); delay(800);
  Serial.println(F("[UJI] Pompa NYALA..."));
  digitalWrite(PIN_RELAI_POMPA,RELAI_NYALA); delay(1500);
  Serial.println(F("[UJI] Pompa MATI"));
  digitalWrite(PIN_RELAI_POMPA,RELAI_MATI);
  lampuNyala=false; statusPompa=POMPA_JEDA; tsPompaBerhenti=millis(); pompaManual=false;
}

const char* namaPompa() {
  switch (statusPompa) {
    case POMPA_SIAP:return "SIAP"; case POMPA_JALAN:return "JALAN";
    case POMPA_JEDA:return "JEDA"; default:return "TERKUNCI";
  }
}

const char* namaPakan() {
  switch (fasePakan) {
    case PAKAN_IDLE: return "SIAP";
    case PAKAN_MAJU:
      if (!servoSudahBuka)  return "JEDA SERVO";
      if (!servoSudahTutup) return "SERVO BUKA";
      return "MOTOR MAJU";
    case PAKAN_TAHAN_UJUNG: return "DI UJUNG";
    case PAKAN_MUNDUR:      return "MOTOR MUNDUR";
    default:                return "TERKUNCI";
  }
}

void barisLcd(uint8_t b, const String& t) {
  if (!lcd) return;
  String s=t;
  if (s.length()>20) s=s.substring(0,20);
  while (s.length()<20) s+=' ';
  lcd->setCursor(0,b); lcd->print(s);
}

String teksWaktu() {
  if (!rtcSehat && !ntpPernahSukses) return "-- JAM BELUM ADA --";
  char b[24];
  snprintf(b,sizeof(b),"%s %02d/%02d %02d:%02d",NAMA_HARI[waktuKini.dayOfTheWeek()],
    waktuKini.day(),waktuKini.month(),waktuKini.hour(),waktuKini.minute());
  return String(b);
}

String teksKoneksi() { return (WiFi.status()==WL_CONNECTED)?"ONLINE":"OFFLINE"; }

void gambarHalaman() {
  if (!lcd) return;
  char b[24];
  switch (halaman) {
    case 0:
      barisLcd(0,teksWaktu());
      if (adaSht31&&!isnan(suhu)) {
        barisLcd(1,"Suhu  : "+String(suhu,1)+" C");
        barisLcd(2,"Lembap: "+String(kelembapan,1)+" %");
      } else { barisLcd(1,"Suhu  : ERR"); barisLcd(2,"Lembap: ERR"); }
      barisLcd(3,"Net   : "+teksKoneksi()+(mqtt.connected()?" MQ":""));
      break;

    case 1:
      barisLcd(0,"== AIR ==");
      barisLcd(1,"Tangki: "+String(namaTangki()));
      barisLcd(2,"Pompa : "+String(namaPompa())+(pompaManual?" MAN":""));
      barisLcd(3,"Pelamp: bwh"+String(pelBawah?1:0)+" ats"+String(pelAtas?1:0)
                 +"  "+String(otoPompa?"OTO":"MAN"));
      break;

    case 2:
      barisLcd(0,"== PAKAN ==");
      barisLcd(1,String(namaPakan()));
      barisLcd(2,"Level : "+String(persenPakan>=0?String(persenPakan)+" %":"--"));
      if (pakanJam<=23) { snprintf(b,sizeof(b),"%02d:%02d",pakanJam,pakanMenit);
        barisLcd(3,"Akhir : "+String(b)+" x"+String(siklusPakan)); }
      else barisLcd(3,"Akhir : belum");
      break;

    case 3:
      barisLcd(0,"== DAYA ==");
      barisLcd(1,"Aki   : "+String(isnan(teganganRata)?"-- ":String(teganganRata,2))+" V");
      barisLcd(2,"SoC   : "+String(estimasiSoc(teganganRata))+" %");
      barisLcd(3,"Status: "+String(statusAki()));
      break;

    case 4:
      barisLcd(0,"== MODE ==");
      barisLcd(1,"Pompa : "+String(otoPompa?"OTOMATIS":"MANUAL"));
      barisLcd(2,"Lampu : "+String(otoLampu?"OTOMATIS":"MANUAL"));
      barisLcd(3,"Pakan : "+String(otoPakan?"OTOMATIS":"MANUAL"));
      break;
  }
}
